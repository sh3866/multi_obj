// --- Math Helpers ---

const Vec3 = {
  add: (v1, v2) => ({ x: v1.x + v2.x, y: v1.y + v2.y, z: v1.z + v2.z }),
  sub: (v1, v2) => ({ x: v1.x - v2.x, y: v1.y - v2.y, z: v1.z - v2.z }),
  dot: (v1, v2) => v1.x * v2.x + v1.y * v2.y + v1.z * v2.z,
  len: (v) => Math.sqrt(v.x*v.x + v.y*v.y + v.z*v.z),
  normalize: (v) => {
    const l = Math.sqrt(v.x*v.x + v.y*v.y + v.z*v.z);
    return l > 0 ? { x: v.x/l, y: v.y/l, z: v.z/l } : {x:0,y:0,z:0};
  },
  scale: (v, s) => ({ x: v.x * s, y: v.y * s, z: v.z * s }),
  distSq: (v1, v2) => {
    const dx = v1.x - v2.x;
    const dy = v1.y - v2.y;
    const dz = v1.z - v2.z;
    return dx*dx + dy*dy + dz*dz;
  }
};

// Fibonacci Lattice Sphere Generation
function createSpherePoints(count, radius) {
  const points = [];
  const goldenAngle = Math.PI * (3 - Math.sqrt(5)); 
  
  for (let i = 0; i < count; i++) {
    const y = 1 - (i / (count - 1)) * 2; 
    const radiusAtY = Math.sqrt(1 - y * y); 
    const theta = goldenAngle * i; 
    
    const x = Math.cos(theta) * radiusAtY;
    const z = Math.sin(theta) * radiusAtY;
    
    points.push({
      x: x * radius,
      y: y * radius,
      z: z * radius,
      idx: i
    });
  }
  return points;
}

// --- Game Logic ---

function initGame() {
  const points = createSpherePoints(CONFIG.cellCount, CONFIG.sphereRadius);
  
  // Neighbor Search Optimization: 
  // Instead of O(N^2), we can bucket by latitude (y-axis) or just do N^2 if N < 5000.
  // For 4500, N^2 is ~20M checks. It might take 100-200ms. Acceptable for init.
  
  // Calculate average distance to nearest neighbor to define "neighbor" threshold
  // Approx area of sphere = 4*pi*r^2. Area per cell = (4*pi*r^2)/N.
  // Radius of cell area ~ sqrt(Area/pi).
  const avgDist = (2 * Math.PI * CONFIG.sphereRadius) / Math.sqrt(CONFIG.cellCount) * 0.8;
  const thresholdSq = avgDist * avgDist;

  state.cells = points.map((p, i) => {
    const neighbors = [];
    // Optimization: Only check points with similar Y to reduce checks? 
    // For simplicity and robustness, we'll do full check but use distanceSq.
    // In a production app, use a grid or k-d tree.
    
    for (let j = 0; j < points.length; j++) {
      if (i === j) continue;
      // Quick reject on Y axis
      if (Math.abs(p.y - points[j].y) > avgDist) continue;
      
      const dSq = Vec3.distSq(p, points[j]);
      if (dSq < thresholdSq) { 
        neighbors.push(j);
      }
    }
    
    return {
      pos: p,
      idx: i,
      neighbors: neighbors,
      alive: Math.random() > 0.75, // 25% initial density
      bornGen: 0, 
      age: 0
    };
  });
  
  state.generation = 0;
  updateStats();
}

function stepGame() {
  const nextStates = state.cells.map(cell => {
    // Count alive neighbors
    let neighborCount = 0;
    for (let k = 0; k < cell.neighbors.length; k++) {
      if (state.cells[cell.neighbors[k]].alive) neighborCount++;
    }
    
    let nextAlive = false;
    let isBorn = false;

    // Conway Rules
    if (cell.alive) {
      if (neighborCount === 2 || neighborCount === 3) nextAlive = true;
    } else {
      if (neighborCount === 3) {
        nextAlive = true;
        isBorn = true;
      }
    }
    
    return {
      alive: nextAlive,
      bornGen: isBorn ? state.generation : cell.bornGen,
      age: nextAlive ? (cell.alive ? cell.age + 1 : 0) : 0
    };
  });

  state.cells.forEach((cell, i) => {
    cell.alive = nextStates[i].alive;
    cell.bornGen = nextStates[i].bornGen;
    cell.age = nextStates[i].age;
  });

  state.generation++;
  updateStats();
}

// --- Rendering ---

function project(p, rotX, rotY, zoom) {
  // Rotate Y
  let x = p.x * Math.cos(rotY) - p.z * Math.sin(rotY);
  let z = p.z * Math.cos(rotY) + p.x * Math.sin(rotY);
  
  // Rotate X
  let y = p.y * Math.cos(rotX) - z * Math.sin(rotX);
  let zFinal = z * Math.cos(rotX) + p.y * Math.sin(rotX);
  
  // Perspective
  const fov = 900 * zoom;
  const scale = fov / (fov + zFinal);
  
  return {
    x: x * scale + width / 2,
    y: y * scale + height / 2,
    scale: scale,
    z: zFinal
  };
}

/**
 * Generates a color based on the cell's age/generation and position.
 * Palette: Cyan (180) -> Violet (280) -> Magenta (300) -> Amber (30)
 */
function getCellColor(bornGen, zDepth) {
  // Base hue oscillates slowly over time for the whole sphere, 
  // but individual cells keep their birth color identity to show patterns.
  
  // Map generation to Hue
  // We want a spread. Let's say 100 generations span the palette.
  const paletteSpan = 150; 
  const relativeGen = bornGen % paletteSpan;
  
  // Map 0..150 to 180 (Cyan) .. 30 (Amber) wrapping around.
  // 180 -> 270 (Blue/Violet) -> 300 (Magenta) -> 30 (Amber) -> 180
  // Range is 150 degrees.
  let hue = 180 + (relativeGen / paletteSpan) * 150;
  if (hue > 360) hue -= 360;
  
  // Amber is around 30-40. If we want to hit Amber specifically, we might need a non-linear map,
  // but 180->330 covers Cyan->Magenta. To get Amber (30), we need to wrap or extend.
  // Let's adjust the map to ensure Amber is hit.
  // 180 (Cyan) -> 260 (Violet) -> 310 (Magenta) -> 30 (Amber) -> 180
  // This is a spread of ~270 degrees.
  
  hue = 180 + (relativeGen / paletteSpan) * 270;
  if (hue > 360) hue -= 360;

  // Lighting / Depth Shading
  // zDepth is positive when facing camera (in our projection logic, zFinal is positive behind? No, usually negative is behind in standard 3D, but here zFinal > 0 is behind center).
  // Let's check projection: scale = fov / (fov + z). If z is positive, scale < 1.
  // So positive z is "far/back". Negative z is "near/front".
  // We want front (negative z) to be bright (100%), back (positive z) to be dim (20%).
  
  // Normalize z to -1..1 approx (using radius)
  const depthFactor = zDepth / CONFIG.sphereRadius; // -1 (front) to 1 (back)
  
  // Lightness calculation: 
  // Front (-1) -> Lightness 65%
  // Back (1) -> Lightness 15%
  const lightness = 65 - (depthFactor * 25); 
  
  // Saturation: High on front, slightly lower on back for "fog" effect
  const saturation = 85 - (depthFactor * 10);

  return `hsl(${hue}, ${saturation}%, ${lightness}%)`;
}

function draw() {
  // Clear background
  ctx.fillStyle = '#050508';
  ctx.fillRect(0, 0, width, height);

  // Draw Halo behind sphere
  const cx = width / 2;
  const cy = height / 2;
  const r = CONFIG.sphereRadius * state.zoom;
  
  // Halo Gradient
  const haloGrad = ctx.createRadialGradient(cx, cy, r * 0.8, cx, cy, r * 1.4);
  haloGrad.addColorStop(0, 'rgba(0, 0, 0, 0)');
  haloGrad.addColorStop(0.5, 'rgba(100, 100, 255, 0.03)'); // Subtle blue tint
  haloGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
  
  ctx.fillStyle = haloGrad;
  ctx.beginPath();
  ctx.arc(cx, cy, r * 1.5, 0, Math.PI * 2);
  ctx.fill();

  // Project Cells
  const projectedCells = state.cells.map(cell => {
    const proj = project(cell.pos, state.rotation.x, state.rotation.y, state.zoom);
    return { cell, proj };
  });
  
  // Sort back-to-front
  projectedCells.sort((a, b) => b.proj.z - a.proj.z);

  // Draw Cells
  projectedCells.forEach(({ cell, proj }) => {
    // Cull strictly back-facing cells to save perf and improve clarity
    // If z > 0 (behind center plane) and scale is small, maybe skip?
    // Let's keep them but make them very dim via color function.
    
    const size = CONFIG.cellSize * proj.scale;
    
    if (cell.alive) {
      const color = getCellColor(cell.bornGen, proj.z);
      ctx.fillStyle = color;
      
      // Add glow to front-facing alive cells
      if (proj.z < 0 && size > 2) {
        ctx.shadowBlur = 6 * proj.scale;
        ctx.shadowColor = color;
      } else {
        ctx.shadowBlur = 0;
      }
      
      ctx.beginPath();
      ctx.arc(proj.x, proj.y, size, 0, Math.PI * 2);
      ctx.fill();
    } else {
      // Dead cells: very faint grid
      // Dim them significantly based on depth
      const alpha = 0.15 - (proj.z / CONFIG.sphereRadius) * 0.1;
      if (alpha > 0.01) {
        ctx.fillStyle = `rgba(100, 110, 130, ${alpha})`;
        ctx.shadowBlur = 0;
        ctx.beginPath();
        ctx.arc(proj.x, proj.y, size * 0.8, 0, Math.PI * 2);
        ctx.fill();
      }
    }
  });
  
  ctx.shadowBlur = 0; // Reset
}

// --- Main Loop ---

function loop(timestamp) {
  if (state.isRunning) {
    // Auto rotation
    state.rotation.y += CONFIG.rotationSpeed;
    
    // Update Logic based on speed
    // Speed slider 1-60. 
    // If speed is 60, update every frame (approx 16ms)
    // If speed is 1, update every 1000ms
    const interval = 1000 / state.speed;
    
    if (timestamp - state.lastFrameTime > interval) {
      stepGame();
      state.lastFrameTime = timestamp;
    }
  }
  
  draw();
  requestAnimationFrame(loop);
}
