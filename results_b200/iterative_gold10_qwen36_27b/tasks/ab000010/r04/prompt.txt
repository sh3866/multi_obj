/**
 * Spherical Conway's Game of Life - Visual Revision
 * 
 * Design Goals:
 * 1. High Density: ~8000 cells for a rich texture.
 * 2. Multicolor: Explicit mapping of cell index (Longitude) to specific palette colors.
 * 3. Depth: Strong Z-shading to create 3D volume.
 * 4. Bloom: Per-color glow effects.
 */

// --- Configuration ---
const CONFIG = {
  cellCount: 8000, // High density
  sphereRadius: 240,
  cellSize: 2.2, // Smaller cells to fit density
  rotationSpeed: 0.001,
  // Explicit Palette
  palette: [
    '#19e6ff', // Cyan
    '#4d7cff', // Blue
    '#8b5cf6', // Violet
    '#ec4899', // Magenta
    '#f6b94a'  // Amber
  ]
};

// --- State ---
let state = {
  cells: [],
  generation: 0,
  isRunning: true,
  speed: 20,
  lastFrameTime: 0,
  rotation: { x: 0.2, y: 0.5 },
  zoom: 1.2,
  isDragging: false,
  lastMouse: { x: 0, y: 0 }
};

// --- Canvas Setup ---
const canvas = document.getElementById('sphereCanvas');
const ctx = canvas.getContext('2d', { alpha: false }); 

let width, height;

function resize() {
  width = window.innerWidth;
  height = window.innerHeight;
  canvas.width = width;
  canvas.height = height;
}
window.addEventListener('resize', resize);
resize();

// --- Math Helpers ---

const Vec3 = {
  distSq: (v1, v2) => {
    const dx = v1.x - v2.x;
    const dy = v1.y - v2.y;
    const dz = v1.z - v2.z;
    return dx*dx + dy*dy + dz*dz;
  }
};

// Fibonacci Lattice Sphere Generation
function createSpherePoints(count, radius) {
  const points = [];
  const goldenAngle = Math.PI * (3 - Math.sqrt(5)); 
  
  for (let i = 0; i < count; i++) {
    const y = 1 - (i / (count - 1)) * 2; 
    const radiusAtY = Math.sqrt(1 - y * y); 
    const theta = goldenAngle * i; 
    
    const x = Math.cos(theta) * radiusAtY;
    const z = Math.sin(theta) * radiusAtY;
    
    points.push({
      x: x * radius,
      y: y * radius,
      z: z * radius,
      idx: i,
      theta: theta // Store theta for deterministic coloring
    });
  }
  return points;
}

// --- Game Logic ---

function initGame() {
  const points = createSpherePoints(CONFIG.cellCount, CONFIG.sphereRadius);
  
  // Calculate neighbor threshold
  // Adjusted for higher density
  const avgDist = (2 * Math.PI * CONFIG.sphereRadius) / Math.sqrt(CONFIG.cellCount) * 0.85;
  const thresholdSq = avgDist * avgDist;

  state.cells = points.map((p, i) => {
    const neighbors = [];
    
    for (let j = 0; j < points.length; j++) {
      if (i === j) continue;
      // Optimization: Y-axis rejection
      if (Math.abs(p.y - points[j].y) > avgDist) continue;
      
      const dSq = Vec3.distSq(p, points[j]);
      if (dSq < thresholdSq) { 
        neighbors.push(j);
      }
    }
    
    // Initialize with dense clusters (High Population Start)
    // Select ~20 random seeds
    const seeds = [];
    for(let s=0; s<20; s++) {
        seeds.push(points[Math.floor(Math.random() * points.length)]);
    }
    
    let isAlive = false;
    // Check distance to seeds
    for(let s of seeds) {
        // Increased cluster radius for density
        if(Vec3.distSq(p, s) < (avgDist * avgDist * 20)) { 
            isAlive = true;
            break;
        }
    }
    
    // Assign color deterministically based on longitude (theta)
    // Map theta (0 to 2PI) to 0-1 range, then to palette index
    const normalizedTheta = (p.theta + Math.PI) / (2 * Math.PI);
    const colorIndex = Math.floor(normalizedTheta * CONFIG.palette.length) % CONFIG.palette.length;
    
    return {
      pos: p,
      idx: i,
      neighbors: neighbors,
      alive: isAlive, 
      age: 0,
      colorHex: CONFIG.palette[colorIndex] // Store hex for easier shadow manipulation
    };
  });
  
  state.generation = 0;
  updateStats();
}

function stepGame() {
  const nextStates = state.cells.map(cell => {
    // Count alive neighbors
    let neighborCount = 0;
    for (let k = 0; k < cell.neighbors.length; k++) {
      if (state.cells[cell.neighbors[k]].alive) neighborCount++;
    }
    
    let nextAlive = false;

    // Conway Rules
    if (cell.alive) {
      if (neighborCount === 2 || neighborCount === 3) nextAlive = true;
    } else {
      if (neighborCount === 3) {
        nextAlive = true;
      }
    }
    
    return {
      alive: nextAlive,
      age: nextAlive ? (cell.alive ? cell.age + 1 : 0) : 0
    };
  });

  state.cells.forEach((cell, i) => {
    cell.alive = nextStates[i].alive;
    cell.age = nextStates[i].age;
  });

  state.generation++;
  updateStats();
}

// --- Rendering ---

function project(p, rotX, rotY, zoom) {
  // Rotate Y
  let x = p.x * Math.cos(rotY) - p.z * Math.sin(rotY);
  let z = p.z * Math.cos(rotY) + p.x * Math.sin(rotY);
  
  // Rotate X
  let y = p.y * Math.cos(rotX) - z * Math.sin(rotX);
  let zFinal = z * Math.cos(rotX) + p.y * Math.sin(rotX);
  
  // Perspective
  const fov = 900 * zoom;
  const scale = fov / (fov + zFinal);
  
  return {
    x: x * scale + width / 2,
    y: y * scale + height / 2,
    scale: scale,
    z: zFinal
  };
}

function hexToRgb(hex) {
  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
  return result ? {
    r: parseInt(result[1], 16),
    g: parseInt(result[2], 16),
    b: parseInt(result[3], 16)
  } : null;
}

function draw() {
  // Clear background
  ctx.fillStyle = '#050508';
  ctx.fillRect(0, 0, width, height);

  // Draw Halo behind sphere
  const cx = width / 2;
  const cy = height / 2;
  const r = CONFIG.sphereRadius * state.zoom;
  
  // Halo Gradient
  const haloGrad = ctx.createRadialGradient(cx, cy, r * 0.8, cx, cy, r * 1.6);
  haloGrad.addColorStop(0, 'rgba(0, 0, 0, 0)');
  haloGrad.addColorStop(0.6, 'rgba(50, 50, 100, 0.08)'); 
  haloGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
  
  ctx.fillStyle = haloGrad;
  ctx.beginPath();
  ctx.arc(cx, cy, r * 1.8, 0, Math.PI * 2);
  ctx.fill();

  // Project Cells
  const projectedCells = state.cells.map(cell => {
    const proj = project(cell.pos, state.rotation.x, state.rotation.y, state.zoom);
    return { cell, proj };
  });
  
  // Sort back-to-front (Painter's Algorithm)
  projectedCells.sort((a, b) => b.proj.z - a.proj.z);

  // Draw Cells
  projectedCells.forEach(({ cell, proj }) => {
    const size = CONFIG.cellSize * proj.scale;
    
    // 1. Lighting / Depth Shading
    // zDepth is positive behind, negative in front.
    const depthFactor = proj.z / CONFIG.sphereRadius; 
    
    // 2. Color Calculation
    if (cell.alive) {
      const baseColor = hexToRgb(cell.colorHex);
      
      // Apply shading based on depth
      // Front (-1) -> 100% opacity/brightness, Back (1) -> 40%
      // Using simple alpha blending for performance + visual depth
      const alpha = 1.0 - (depthFactor * 0.6); // 1.0 to 0.4
      const lightMod = 1.0 - (depthFactor * 0.3); // 1.0 to 0.7

      const r = Math.min(255, baseColor.r * lightMod);
      const g = Math.min(255, baseColor.g * lightMod);
      const b = Math.min(255, baseColor.b * lightMod);
      
      ctx.fillStyle = `rgba(${r},${g},${b},${alpha})`;
      
      // Bloom/Glow: Only apply to front-facing alive cells to save performance
      // And only if cell is large enough on screen
      if (proj.z < -50 && size > 1.5) {
        ctx.shadowBlur = 6 * proj.scale;
        ctx.shadowColor = cell.colorHex;
      } else {
        ctx.shadowBlur = 0;
      }
      
      ctx.beginPath();
      ctx.arc(proj.x, proj.y, size, 0, Math.PI * 2);
      ctx.fill();
    } else {
      // Dead cells: faint grid dots
      // Only draw if on the front face to save perf
      if (proj.z < CONFIG.sphereRadius * 0.8) {
        const alpha = 0.15 - (proj.z / CONFIG.sphereRadius) * 0.08;
        if (alpha > 0.01) {
          ctx.fillStyle = `rgba(100, 110, 130, ${alpha})`;
          ctx.shadowBlur = 0;
          ctx.beginPath();
          ctx.arc(proj.x, proj.y, size * 0.6, 0, Math.PI * 2);
          ctx.fill();
        }
      }
    }
  });
  
  ctx.shadowBlur = 0; // Reset
}

// --- Main Loop ---

function loop(timestamp) {
  if (state.isRunning) {
    // Auto rotation
    state.rotation.y += CONFIG.rotationSpeed;
    
    // Update Logic based on speed
    const interval = 1000 / state.speed;
    
    if (timestamp - state.lastFrameTime > interval) {
      stepGame();
      state.lastFrameTime = timestamp;
    }
  }
  
  draw();
  requestAnimationFrame(loop);
}

}
