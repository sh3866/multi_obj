    // --- Background Particle System ---
    const canvas = document.getElementById('canvas-bg');
    const ctx = canvas.getContext('2d');

    let width, height;
    let particles = [];

    function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;
    }

    class Particle {
        constructor() {
            this.reset();
        }

        reset() {
            this.x = Math.random() * width;
            this.y = height + Math.random() * 100;
            this.vx = (Math.random() - 0.5) * 0.2; // Slower drift
            this.vy = -(Math.random() * 0.5 + 0.2); // Slower rise
            this.size = Math.random() * 1.5;
            this.alpha = Math.random() * 0.4;
            this.life = Math.random() * 0.5 + 0.5;
            // More gold particles to match identity
            this.color = Math.random() > 0.6 ? '212, 175, 55' : '79, 209, 197'; 
        }

        update() {
            this.x += this.vx;
            this.y += this.vy;
            this.alpha -= 0.001;

            if (this.alpha <= 0 || this.y < -10) {
                this.reset();
            }
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(${this.color}, ${this.alpha})`;
            ctx.fill();
        }
    }

    function initParticles() {
        particles = [];
        // Reduce count for cleaner look
        const count = Math.floor((width * height) / 25000); 
        for (let i = 0; i < count; i++) {
            particles.push(new Particle());
        }
    }

    function animate() {
        ctx.clearRect(0, 0, width, height);
        
        for (let i = 0; i < particles.length; i++) {
            particles[i].update();
            particles[i].draw();
        }
        requestAnimationFrame(animate);
    }

    // --- Interactions ---
    window.addEventListener('resize', () => {
        resize();
        initParticles();
    });

    function interact(action) {
        const btn = event.currentTarget;
        // Visual feedback handled by CSS active/hover, but we can add a small log
        console.log(`Action triggered: ${action}`);
    }

    // Initialize
    resize();
    initParticles();
    animate();

</script>
