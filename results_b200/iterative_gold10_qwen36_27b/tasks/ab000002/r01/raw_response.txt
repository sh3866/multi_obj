    function getStatus(skill) {
        if (skill.unlocked) return 'unlocked';
        
        // Check if parent is unlocked
        if (!skill.req) return 'locked'; // Root is special
        
        const parent = skills.find(s => s.id === skill.req);
        if (parent && parent.unlocked) return 'available';
        
        return 'locked';
    }
