    function getStatus(skill) {
        if (skill.unlocked) return 'unlocked';
        
        // Check if parent is unlocked
        if (!skill.req) return 'locked'; // Root is special
        
        const parent = skills.find(s => s.id === skill.req);
        if (parent && parent.unlocked) return 'available';
        
        return 'locked';
    }

    function drawLines() {
        svg.innerHTML = ''; // Clear existing
        const containerRect = container.getBoundingClientRect();
        
        // Helper to convert % to absolute px relative to container
        const getPos = (pctX, pctY) => ({
            x: (pctX / 100) * containerRect.width,
            y: (pctY / 100) * containerRect.height
        });

        skills.forEach(skill => {
            if (skill.req) {
                const parent = skills.find(s => s.id === skill.req);
                if (parent) {
                    const pPos = getPos(parent.x, parent.y);
                    const cPos = getPos(skill.x, skill.y);
                    
                    const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                    line.setAttribute('x1', pPos.x);
                    line.setAttribute('y1', pPos.y);
                    line.setAttribute('x2', cPos.x);
                    line.setAttribute('y2', cPos.y);
                    
                    // Determine line style
                    let lineClass = 'connection-line';
                    if (skill.unlocked) lineClass += ' active';
                    else if (parent.unlocked) lineClass += ' available'; // Parent unlocked, child not
                    
                    line.setAttribute('class', lineClass);
                    svg.appendChild(line);
                }
            }
        });
    }

    function showTooltip(e, skill) {

    }

        const status = getStatus(skill);
        if (status === 'available') {
            skill.unlocked = true;
            init(); // Re-render to update states and lines
            hideTooltip();
        }
    }
