<script>
  /**
   * Game Logic & Rendering
   */
  
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  
  // CRITIQUE FIX: Fixed initial positions for Player and Net
  // Player at 980/620, Net at 850/540
  let playerPos = { x: 980, y: 620 };
  let mouse = { x: 850, y: 540 }; // Net starts here
  let isMouseDown = false;
  let time = 0;

  const COLORS = ['#00f0ff', '#ff0055', '#ccff00', '#bd00ff'];
  const CREATURE_COUNT = 5; 

  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
    
    // CRITIQUE FIX: Clamp player/net to safe margins if screen is small
    // Safe margin 40px
    playerPos.x = Math.min(playerPos.x, width - 40);
    playerPos.y = Math.min(playerPos.y, height - 40);
    mouse.x = Math.min(mouse.x, width - 40);
    mouse.y = Math.min(mouse.y, height - 40);
  }
  window.addEventListener('resize', resize);
  resize();

  window.addEventListener('mousemove', e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
  });
  window.addEventListener('mousedown', () => isMouseDown = true);
  window.addEventListener('mouseup', () => {
    isMouseDown = false;
    checkCatch();
  });
  window.addEventListener('touchmove', e => {
    e.preventDefault(); 
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
    isMouseDown = true;
  });
  window.addEventListener('touchend', () => {
    isMouseDown = false;
    checkCatch();
  });

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 4 + 1;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.02 + 0.01;
      this.size = Math.random() * 4 + 2;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.life -= this.decay;
      this.size *= 0.96;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor(startX, startY, type = 'random') {
      this.x = startX;
      this.y = startY;
      this.vx = (Math.random() - 0.5) * 1.5;
      this.vy = (Math.random() - 0.5) * 1.5;
      this.radius = Math.random() * 25 + 20;
      
      if (type === 'yellow') this.color = '#ccff00';
      else if (type === 'cyan') this.color = '#00f0ff';
      else if (type === 'magenta') this.color = '#ff0055';
      else this.color = COLORS[Math.floor(Math.random() * COLORS.length)];
      
      this.angle = 0;
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = Math.random() * 0.03 + 0.01;
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      this.x += this.vx;
      this.y += this.vy;
      this.wobble += this.wobbleSpeed;
      
      this.x += Math.sin(this.wobble) * 0.3;
      this.y += Math.cos(this.wobble * 0.8) * 0.3;

      // CRITIQUE FIX: Constrain to safe area (40px margin)
      const minX = 40;
      const maxX = width - 40;
      const minY = 40;
      const maxY = height - 40;

      if (this.x < minX || this.x > maxX) this.vx *= -1;
      if (this.y < minY || this.y > maxY) this.vy *= -1;

      // Hard clamp
      this.x = Math.max(minX, Math.min(this.x, maxX));
      this.y = Math.max(minY, Math.min(this.y, maxY));

      // Avoid Mouse (Net)
      const dx = this.x - mouse.x;
      const dy = this.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      if (dist < 200) {
        const force = (200 - dist) / 200;
        this.vx += (dx / dist) * force * 0.8;
        this.vy += (dy / dist) * force * 0.8;
      }

      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 3) {
        this.vx = (this.vx / speed) * 3;
        this.vy = (this.vy / speed) * 3;
      }
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      
      ctx.shadowBlur = 30;
      ctx.shadowColor = this.color;
      
      ctx.fillStyle = this.color;
      ctx.beginPath();
      
      const r = this.radius;
      const wobbleOffset = Math.sin(this.wobble * 2) * 5;
      
      ctx.moveTo(r + wobbleOffset, 0);
      ctx.bezierCurveTo(r, r, 0, r, -r - wobbleOffset, 0);
      ctx.bezierCurveTo(-r, -r, 0, -r, r + wobbleOffset, 0);
      
      ctx.fill();

      ctx.fillStyle = '#fff';
      ctx.shadowBlur = 0;
      ctx.beginPath();
      ctx.arc(0, 0, r * 0.4, 0, Math.PI * 2);
      ctx.fill();

      ctx.strokeStyle = this.color;
      ctx.lineWidth = 2;
      ctx.globalAlpha = 0.6;
      ctx.beginPath();
      ctx.moveTo(0, r);
      ctx.quadraticCurveTo(Math.sin(this.wobble)*20, r+20, Math.sin(this.wobble)*10, r+30);
      ctx.stroke();
      
      ctx.restore();
    }
  }

  function init() {
    creatures = [];
    score = 0;
    scoreEl.innerText = score;

    // CRITIQUE FIX: Explicitly place 5 creatures at fixed visible positions
    // 1. Cyan Upper Left
    creatures.push(new Creature(150, 180, 'cyan'));
    // 2. Magenta Upper Left (separated)
    creatures.push(new Creature(280, 150, 'magenta'));
    // 3. Yellow Center-Left
    creatures.push(new Creature(400, 300, 'yellow'));
    // 4. Purple Mid-Right
    creatures.push(new Creature(700, 250, 'purple'));
    // 5. Cyan Mid-Right
    creatures.push(new Creature(800, 450, 'cyan'));
  }

  function checkCatch() {
    const catchRadius = 80;
    let caughtSomething = false;
    creatures.forEach(c => {
      if (c.isCaught) return;
      const dx = c.x - mouse.x;
      const dy = c.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      if (dist < catchRadius) {
        catchCreature(c);
        caughtSomething = true;
      }
    });
    if (!caughtSomething) createSwipeEffect();
  }
  
  function createSwipeEffect() {
      for(let i=0; i<5; i++) {
        let p = new Particle(mouse.x, mouse.y, '#ffffff');
        p.color = 'rgba(255,255,255,0.5)';
        p.decay = 0.05;
        particles.push(p);
      }
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    for(let i=0; i<25; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }
    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1200);
    setTimeout(() => { 
      // Respawn somewhere safe
      creature.x = 100 + Math.random() * (width - 200);
      creature.y = 100 + Math.random() * (height - 200);
      creature.isCaught = false;
    }, 3000);
  }

  function resetGame() {
    init();
  }

  function drawPlayerAndNet() {
    // CRITIQUE FIX: Player at 980/620, Net at 850/540
    
    // 1. Player Silhouette
    ctx.save();
    ctx.translate(playerPos.x, playerPos.y);
    
    ctx.fillStyle = '#050508';
    ctx.shadowColor = 'rgba(0, 240, 255, 0.3)'; 
    ctx.shadowBlur = 20; 
    
    ctx.beginPath();
    // Head
    ctx.arc(0, -110, 18, 0, Math.PI*2);
    ctx.fill();

    // Torso
    ctx.beginPath();
    ctx.moveTo(0, -92); 
    ctx.lineTo(25, -80); 
    ctx.lineTo(30, 0); 
    ctx.lineTo(20, 100); 
    ctx.lineTo(-20, 100); 
    ctx.lineTo(-30, 0); 
    ctx.lineTo(-25, -80); 
    ctx.closePath();
    ctx.fill();
    
    // Arm reaching out to Net
    ctx.lineWidth = 14; 
    ctx.strokeStyle = '#050508';
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';
    
    const dx = mouse.x - playerPos.x;
    const dy = mouse.y - playerPos.y;
    const angle = Math.atan2(dy, dx);
    
    // Arm connects shoulder to hand
    ctx.beginPath();
    ctx.moveTo(20, -70); // Shoulder
    ctx.lineTo(mouse.x - playerPos.x + Math.cos(angle)*15, mouse.y - playerPos.y + Math.sin(angle)*15); // Hand position near net
    ctx.stroke();

    ctx.restore();

    // 2. The Net
    ctx.save();
    ctx.translate(mouse.x, mouse.y);
    
    ctx.shadowBlur = 15;
    ctx.shadowColor = isMouseDown ? '#fff' : '#00f0ff';
    
    // Net Rim
    ctx.strokeStyle = isMouseDown ? '#fff' : '#00f0ff';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.arc(0, 0, 70, 0, Math.PI * 2);
    ctx.stroke();
    
    // Net Grid
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.2)';
    ctx.lineWidth = 1;
    for(let i=0; i<6; i++) {
        ctx.beginPath();
        ctx.moveTo(0,0);
        ctx.lineTo(Math.cos(i * Math.PI / 3) * 70, Math.sin(i * Math.PI / 3) * 70);
        ctx.stroke();
    }
    ctx.beginPath(); ctx.arc(0,0, 20, 0, Math.PI*2); ctx.stroke();
    ctx.beginPath(); ctx.arc(0,0, 40, 0, Math.PI*2); ctx.stroke();
    ctx.beginPath(); ctx.arc(0,0, 60, 0, Math.PI*2); ctx.stroke();

    // Net Glow
    const netGlow = ctx.createRadialGradient(0, 0, 10, 0, 0, 65);
    netGlow.addColorStop(0, 'rgba(255, 255, 255, 0.15)');
    netGlow.addColorStop(0.5, 'rgba(0, 240, 255, 0.05)');
    netGlow.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = netGlow;
    ctx.beginPath();
    ctx.arc(0, 0, 70, 0, Math.PI * 2);
    ctx.fill();

    // Handle Connection
    ctx.strokeStyle = '#222';
    ctx.lineWidth = 4;
    ctx.beginPath();
    ctx.moveTo(0,0);
    const handleAngle = Math.atan2(playerPos.y - mouse.y, playerPos.x - mouse.x);
    const handleLen = 60; 
    ctx.lineTo(Math.cos(handleAngle) * handleLen, Math.sin(handleAngle) * handleLen);
    ctx.stroke();
    
    // Handle Grip
    ctx.fillStyle = '#111';
    ctx.beginPath();
    ctx.arc(Math.cos(handleAngle) * handleLen, Math.sin(handleAngle) * handleLen, 6, 0, Math.PI*2);
    ctx.fill();

    ctx.fillStyle = '#050508';
    ctx.shadowColor = 'rgba(0, 240, 255, 0.5)';
    ctx.shadowBlur = 10;
    ctx.beginPath();
    ctx.arc(Math.cos(handleAngle) * handleLen, Math.sin(handleAngle) * handleLen, 9, 0, Math.PI*2);
    ctx.fill();

    ctx.restore();
  }

  function drawTerrain() {
    ctx.save();
    
    // Layer 1 (Far)
    ctx.fillStyle = '#080812';
    ctx.beginPath();
    ctx.moveTo(0, height);
    ctx.lineTo(0, height - 100);
    for(let i=0; i<=width; i+=50) {
        ctx.lineTo(i, height - 100 + Math.sin(i*0.01 + time*0.5)*20);
    }
    ctx.lineTo(width, height);
    ctx.fill();

    // Layer 2 (Mid)
    ctx.fillStyle = '#030306';
    ctx.beginPath();
    ctx.moveTo(0, height);
    ctx.lineTo(0, height - 50);
    for(let i=0; i<=width; i+=20) {
        let h = Math.sin(i*0.05 + time*0.2) * 15 + Math.cos(i*0.02) * 20;
        ctx.lineTo(i, height - 50 - h);
    }
    ctx.lineTo(width, height);
    ctx.fill();
    
    // Rim light
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.1)';
    ctx.lineWidth = 2;
    ctx.stroke();

    // Layer 3: Foreground Reeds (Depth)
    ctx.fillStyle = '#010102';
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.15)';
    ctx.lineWidth = 1;
    
    const reedCount = 15;
    for(let i=0; i<reedCount; i++) {
        const rx = (width / reedCount) * i + Math.sin(i*5)*20;
        const rh = 80 + Math.sin
