<script>
  /**
   * Game Logic & Rendering
   * Pure Canvas implementation for smooth 60fps rendering.
   */
  
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  let mouse = { x: window.innerWidth / 2, y: window.innerHeight - 200 };
  let isMouseDown = false;
  let time = 0;

  // Configuration
  const COLORS = ['#00f0ff', '#ff0055', '#ccff00', '#bd00ff'];
  const CREATURE_COUNT = 5;

  // Resize Handling
  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
    // Reposition mouse if off screen
    if (mouse.x > width) mouse.x = width / 2;
    if (mouse.y > height) mouse.y = height / 2;
  }
  window.addEventListener('resize', resize);
  resize();

  // Input Handling
  window.addEventListener('mousemove', e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
  });
  window.addEventListener('mousedown', () => isMouseDown = true);
  window.addEventListener('mouseup', () => {
    isMouseDown = false;
    checkCatch();
  });
  // Touch support
  window.addEventListener('touchmove', e => {
    e.preventDefault(); 
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
    isMouseDown = true;
  });
  window.addEventListener('touchend', () => {
    isMouseDown = false;
    checkCatch();
  });

  // --- Classes ---

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 4 + 1;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.02 + 0.01;
      this.size = Math.random() * 4 + 2;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.life -= this.decay;
      this.size *= 0.95;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor(startNearNet = false) {
      this.reset(startNearNet);
      // Start at random positions initially if not near net
      if (!startNearNet) {
        this.x = Math.random() * width;
        this.y = Math.random() * (height * 0.6); // Keep in upper area
      }
    }

    reset(startNearNet = false) {
      // Spawn away from center
      this.x = startNearNet ? mouse.x + 150 : (Math.random() < 0.5 ? -100 : width + 100);
      this.y = startNearNet ? mouse.y - 50 : Math.random() * (height * 0.6);
      
      this.vx = (Math.random() - 0.5) * 1.5;
      this.vy = (Math.random() - 0.5) * 1.5;
      this.radius = Math.random() * 25 + 20;
      this.color = COLORS[Math.floor(Math.random() * COLORS.length)];
      this.angle = 0;
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = Math.random() * 0.03 + 0.01;
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      // Movement
      this.x += this.vx;
      this.y += this.vy;
      this.wobble += this.wobbleSpeed;
      
      // Organic drift
      this.x += Math.sin(this.wobble) * 0.3;
      this.y += Math.cos(this.wobble * 0.8) * 0.3;

      // Bounce off walls (softly)
      if (this.x < 0 || this.x > width) this.vx *= -1;
      if (this.y < 0 || this.y > height * 0.7) this.vy *= -1; // Keep mostly in upper view

      // Avoid Mouse (Net) - flee if too close
      const dx = this.x - mouse.x;
      const dy = this.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      if (dist < 200) {
        // Flee behavior
        const force = (200 - dist) / 200;
        this.vx += (dx / dist) * force * 0.8;
        this.vy += (dy / dist) * force * 0.8;
      }

      // Speed limiter
      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 3) {
        this.vx = (this.vx / speed) * 3;
        this.vy = (this.vy / speed) * 3;
      }
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      
      // Intense Glow
      ctx.shadowBlur = 30;
      ctx.shadowColor = this.color;
      
      // Body
      ctx.fillStyle = this.color;
      ctx.beginPath();
      
      // Draw a slightly organic blob shape
      const r = this.radius;
      const wobbleOffset = Math.sin(this.wobble * 2) * 5;
      
      ctx.moveTo(r + wobbleOffset, 0);
      ctx.bezierCurveTo(r, r, 0, r, -r - wobbleOffset, 0);
      ctx.bezierCurveTo(-r, -r, 0, -r, r + wobbleOffset, 0);
      
      ctx.fill();

      // Inner Core (Bright white center)
      ctx.fillStyle = '#fff';
      ctx.shadowBlur = 0;
      ctx.beginPath();
      ctx.arc(0, 0, r * 0.4, 0, Math.PI * 2);
      ctx.fill();

      // "Tentacles" or trailing bits
      ctx.strokeStyle = this.color;
      ctx.lineWidth = 2;
      ctx.globalAlpha = 0.6;
      ctx.beginPath();
      ctx.moveTo(0, r);
      ctx.quadraticCurveTo(Math.sin(this.wobble)*20, r+20, Math.sin(this.wobble)*10, r+30);
      ctx.stroke();
      
      ctx.restore();
    }
  }

  // --- Game Functions ---

  function init() {
    creatures = [];
    // Ensure one creature starts near the player for the "visual relationship"
    creatures.push(new Creature(true));
    for(let i=0; i<CREATURE_COUNT-1; i++) {
      creatures.push(new Creature(false));
    }
    score = 0;
    scoreEl.innerText = score;
  }

  function checkCatch() {
    // Simple collision detection with the "Net" radius
    const catchRadius = 80; // Size of the net capture zone

    let caughtSomething = false;
    creatures.forEach(c => {
      if (c.isCaught) return;
      
      const dx = c.x - mouse.x;
      const dy = c.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);

      if (dist < catchRadius) {
        catchCreature(c);
        caughtSomething = true;
      }
    });
    
    if (!caughtSomething) {
        // Empty swipe effect
        createSwipeEffect();
    }
  }
  
  function createSwipeEffect() {
      // Create a few faint particles around the net to show movement
      for(let i=0; i<5; i++) {
        let p = new Particle(mouse.x, mouse.y, '#ffffff');
        p.color = 'rgba(255,255,255,0.5)';
        p.decay = 0.05;
        particles.push(p);
      }
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    
    // Visual FX
    for(let i=0; i<25; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }

    // Show Toast
    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1200);

    // Respawn logic (delayed)
    setTimeout(() => {
      creature.reset();
    }, 3000);
  }

  function resetGame() {
    init();
  }

  function drawPlayerAndNet() {
    // We draw the player silhouette and the net in the foreground
    // The "Player" is a silhouette in the bottom right/center
    
    const playerX = width * 0.75; // Position player in lower right foreground
    const playerY = height * 0.85;
    
    // 1. Draw Player Silhouette (Static in scene)
    ctx.save();
    ctx.translate(playerX, playerY);
    
    // Silhouette Color
    ctx.fillStyle = '#050508';
    ctx.shadowColor = 'rgba(0, 240, 255, 0.3)';
    ctx.shadowBlur = 20; // Rim light
    
    // Draw simple humanoid shape
    ctx.beginPath();
    // Head
    ctx.arc(0, -80, 20, 0, Math.PI*2);
    // Torso
    ctx.moveTo(0, -60);
    ctx.lineTo(30, 0); // Right shoulder
    ctx.lineTo(20, 100); // Right hip
    ctx.lineTo(-20, 100); // Left hip
    ctx.lineTo(-30, 0); // Left shoulder
    ctx.fill();
    
    // Arm reaching out towards mouse
    ctx.lineWidth = 12;
    ctx.strokeStyle = '#050508';
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(30, -10); // Shoulder
    
    // Calculate angle to mouse
    const dx = mouse.x - playerX;
    const dy = mouse.y - playerY;
    const angle = Math.atan2(dy, dx);
    
    // Arm joint
    const armLen = 80;
    const elbowX = Math.cos(angle) * armLen;
    const elbowY = Math.sin(angle) * armLen;
    
    ctx.lineTo(elbowX, elbowY);
    ctx.stroke();
    
    // Forearm to Net
    ctx.beginPath();
    ctx.moveTo(elbowX, elbowY);
    ctx.lineTo(mouse.x - playerX, mouse.y - playerY);
    ctx.stroke();
    
    ctx.restore();

    // 2. Draw The Net (Follows Mouse)
    ctx.save();
    ctx.translate(mouse.x, mouse.y);
    
    // Net Glow
    ctx.shadowBlur = 15;
    ctx.shadowColor = isMouseDown ? '#fff' : '#00f0ff';
    
    // Net Shape (Circle with dashed rim)
    ctx.strokeStyle = isMouseDown ? '#fff' : '#00f0ff';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.arc(0, 0, 70, 0, Math.PI * 2);
    ctx.stroke();
    
    // Net Grid (Spider web style)
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.2)';
    ctx.lineWidth = 1;
    for(let i=0; i<6; i++) {
        ctx.beginPath();
        ctx.moveTo(0,0);
        ctx.lineTo(Math.cos(i * Math.PI / 3) * 70, Math.sin(i * Math.PI / 3) * 70);
        ctx.stroke();
    }
    // Rings
    ctx.beginPath(); ctx.arc(0,0, 20, 0, Math.PI*2); ctx.stroke();
    ctx.beginPath(); ctx.arc(0,0, 40, 0, Math.PI*2); ctx.stroke();
    ctx.beginPath(); ctx.arc(0,0, 60, 0, Math.PI*2); ctx.stroke();

    // Handle connection point
    ctx.fillStyle = '#111';
    ctx.beginPath();
    ctx.arc(0,0, 5, 0, Math.PI*2);
    ctx.fill();

    ctx.restore();
  }

  function drawTerrain() {
    // Draw layered silhouettes at the bottom to create depth
    ctx.save();
    
    // Layer 1 (Far)
    ctx.fillStyle = '#080812';
    ctx.beginPath();
    ctx.moveTo(0, height);
    ctx.lineTo(0, height - 100);
    for(let i=0; i<=width; i+=50) {
        ctx.lineTo(i, height - 100 + Math.sin(i*0.01 + time*0.5)*20);
    }
    ctx.lineTo(width, height);
    ctx.fill();

    // Layer 2 (Mid) - Foliage shapes
    ctx.fillStyle = '#030306';
    ctx.beginPath();
    ctx.moveTo(0, height);
    ctx.lineTo(0, height - 50);
    // Jagged foliage
    for(let i=0; i<=width; i+=20) {
        let h = Math.sin(i*0.05 + time*0.2) * 15 + Math.cos(i*0.02) * 20;
        ctx.lineTo(i, height - 50 - h);
    }
    ctx.lineTo(width, height);
    ctx.fill();
    
    // Rim light on terrain
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.1)';
    ctx.lineWidth = 2;
    ctx.stroke();

    ctx.restore();
  }

  function loop() {
    time += 0.01;
    
    // Clear
    ctx.clearRect(0, 0, width, height);

    // Update & Draw Creatures (Behind terrain/player)
    creatures.forEach(c => {
      c.update();
      c.draw(ctx);
    });

    // Draw Terrain (Foreground layers)
    drawTerrain();

    // Update & Draw Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      let p = particles[i];
      p.update();
      p.draw(ctx);
      if (p.life <= 0) particles.splice(i, 1);
    }

    // Draw Player & Net (Top Layer)
    drawPlayerAndNet();

    requestAnimationFrame(loop);
  }

  // Start
  init();
  loop();

</script>
