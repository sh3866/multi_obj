  /**

  const canvas = document.getElementById('gameCanvas');

  const COLORS = ['#00f0ff', '#ff0055', '#ccff00', '#bd00ff'];
  const CREATURE_COUNT = 4; 

  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
    // Keep mouse in bounds on resize
    mouse.x = Math.min(mouse.x, width);
    mouse.y = Math.min(mouse.y, height);
  }
  window.addEventListener('resize', resize);
  resize();

  window.addEventListener('mousemove', e => {

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 4 + 1;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.02 + 0.01;
      this.size = Math.random() * 4 + 2;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.life -= this.decay;
      this.size *= 0.95;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor(startNearNet = false) {
      this.reset(startNearNet);
    }

    reset(startNearNet = false) {
      // CRITIQUE FIX: Hard constraints for creature centers
      // x between 60 and 1220, y between 100 and 680
      const minX = 60;
      const maxX = Math.min(1220, width - 60); // Respect right side of screen if smaller than 1280
      const minY = 100;
      const maxY = Math.min(680, height - 100);

      if (startNearNet) {
        // Start near center/mid to ensure they are visible
        // Ensure center is within bounds
        this.x = width/2 + (Math.random() * 200 - 100);
        this.y = height/2 + (Math.random() * 200 - 100);
      } else {
        this.x = minX + Math.random() * (maxX - minX);
        this.y = minY + Math.random() * (maxY - minY);
      }
      
      this.vx = (Math.random() - 0.5) * 1.5;
      this.vy = (Math.random() - 0.5) * 1.5;
      this.radius = Math.random() * 25 + 20;
      this.color = COLORS[Math.floor(Math.random() * COLORS.length)];
      this.angle = 0;
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = Math.random() * 0.03 + 0.01;
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      this.x += this.vx;
      this.y += this.vy;
      this.wobble += this.wobbleSpeed;
      
      this.x += Math.sin(this.wobble) * 0.3;
      this.y += Math.cos(this.wobble * 0.8) * 0.3;

      // CRITIQUE FIX: Constrain to safe area so they don't clip edges
      // x=60 to x=1220 (or width-60), y=100 to y=680 (or height-100)
      const minX = 60;
      const maxX = Math.min(1220, width - 60);
      const minY = 100;
      const maxY = Math.min(680, height - 100);

      if (this.x < minX || this.x > maxX) this.vx *= -1;
      if (this.y < minY || this.y > maxY) this.vy *= -1;

      // Hard clamp to prevent visual clipping
      this.x = Math.max(minX, Math.min(this.x, maxX));
      this.y = Math.max(minY, Math.min(this.y, maxY));

      // Avoid Mouse
      const dx = this.x - mouse.x;
      const dy = this.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      if (dist < 200) {
        const force = (200 - dist) / 200;
        this.vx += (dx / dist) * force * 0.8;
        this.vy += (dy / dist) * force * 0.8;
      }

      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 3) {
        this.vx = (this.vx / speed) * 3;
        this.vy = (this.vy / speed) * 3;
      }
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      
      ctx.shadowBlur = 30;
      ctx.shadowColor = this.color;
      
      ctx.fillStyle = this.color;
      ctx.beginPath();
      
      const r = this.radius;
      const wobbleOffset = Math.sin(this.wobble * 2) * 5;
      
      ctx.moveTo(r + wobbleOffset, 0);
      ctx.bezierCurveTo(r, r, 0, r, -r - wobbleOffset, 0);
      ctx.bezierCurveTo(-r, -r, 0, -r, r + wobbleOffset, 0);
      
      ctx.fill();

      ctx.fillStyle = '#fff';
      ctx.shadowBlur = 0;
      ctx.beginPath();
      ctx.arc(0, 0, r * 0.4, 0, Math.PI * 2);
      ctx.fill();

      ctx.strokeStyle = this.color;
      ctx.lineWidth = 2;
      ctx.globalAlpha = 0.6;
      ctx.beginPath();
      ctx.moveTo(0, r);
      ctx.quadraticCurveTo(Math.sin(this.wobble)*20, r+20, Math.sin(this.wobble)*10, r+30);
      ctx.stroke();
      
      ctx.restore();
    }
  }

  function init() {
    creatures = [];
    creatures.push(new Creature(true));
    for(let i=0; i<CREATURE_COUNT-1; i++) {
      creatures.push(new Creature(false));
    }
    score = 0;
    scoreEl.innerText = score;
  }

  function checkCatch() {
    const catchRadius = 80;
    let caughtSomething = false;
    creatures.forEach(c => {
      if (c.isCaught) return;
      const dx = c.x - mouse.x;
      const dy = c.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      if (dist < catchRadius) {
        catchCreature(c);
        caughtSomething = true;
      }
    });
    if (!caughtSomething) createSwipeEffect();
  }
  
  function createSwipeEffect() {
      for(let i=0; i<5; i++) {
        let p = new Particle(mouse.x, mouse.y, '#ffffff');
        p.color = 'rgba(255,255,255,0.5)';
        p.decay = 0.05;
        particles.push(p);
      }
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    for(let i=0; i<25; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }
    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1200);
    setTimeout(() => { creature.reset(); }, 3000);
  }

  function resetGame() {
    init();
  }

  function drawPlayerAndNet() {

        ctx.stroke();

    ctx.beginPath(); ctx.arc(0,0, 40, 0, Math.PI*2); ctx.stroke();
    ctx.beginPath(); ctx.arc(0,0, 60, 0, Math.PI*2); ctx.stroke();

    ctx.strokeStyle = '#222';
    ctx.lineWidth = 4;
    ctx.beginPath();

  function drawTerrain() {

    // Layer 1 (Far)
    ctx.fillStyle = '#080812';
    ctx.beginPath();
    ctx.moveTo(0, height);
    ctx.lineTo(0, height - 100);

    ctx.beginPath();
    ctx.moveTo(0, height);
    ctx.lineTo(0, height - 50);

        let h = Math.sin(i*0.05 + time*0.2) * 15 + Math.cos(i*0.02) * 20;
        ctx.lineTo(i, height - 50 - h);
    }
    ctx.lineTo(width, height);
    ctx.fill();
    
    // Rim light
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.1)';
    ctx.lineWidth = 2;

    ctx.strokeStyle = 'rgba(0, 240, 255, 0.15)';
    ctx.lineWidth = 1;

        ctx.stroke();

    ctx.restore();
  }

      p.update();
