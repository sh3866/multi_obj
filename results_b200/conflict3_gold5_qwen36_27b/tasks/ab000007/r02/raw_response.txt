<script>
  /**
   * Game Logic & Rendering
   */
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  
  // Mouse/Touch state
  let input = { x: 0, y: 0, active: false };
  
  // Player Position (Fixed lower-left)
  let player = { x: 0, y: 0 };
  let frameCount = 0;

  const COLORS = {
    cyan: '#00f0ff',
    magenta: '#ff0055',
    lime: '#ccff00',
    deepPurple: '#8800ff'
  };
  const CREATURE_COUNT = 8;

  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
    
    // Position player fixed at lower center-left
    player.x = width * 0.25; 
    player.y = height * 0.82;
    
    // Initialize input to player pos
    input.x = player.x;
    input.y = player.y;
  }
  window.addEventListener('resize', resize);
  resize();

  // Input Handling
  window.addEventListener('mousemove', e => {
    input.x = e.clientX;
    input.y = e.clientY;
  });
  window.addEventListener('mousedown', () => input.active = true);
  window.addEventListener('mouseup', () => {
    input.active = false;
    checkCatch();
  });
  
  window.addEventListener('touchmove', e => {
    e.preventDefault();
    input.x = e.touches[0].clientX;
    input.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    input.x = e.touches[0].clientX;
    input.y = e.touches[0].clientY;
    input.active = true;
  });
  window.addEventListener('touchend', () => {
    input.active = false;
    checkCatch();
  });

  // --- Classes ---

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 5 + 2;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.03 + 0.02;
      this.size = Math.random() * 4 + 1;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.vx *= 0.92;
      this.vy *= 0.92;
      this.life -= this.decay;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor() {
      this.reset(true);
    }

    reset(initial = false) {
      // Spawn logic
      if (initial) {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
      } else {
        this.x = Math.random() < 0.5 ? -100 : width + 100;
        this.y = Math.random() * (height - 100);
      }
      
      this.vx = (Math.random() - 0.5) * 1;
      this.vy = (Math.random() - 0.5) * 1;
      
      // Species & Depth
      const speciesRoll = Math.random();
      
      // Depth factor: 0.5 (far) to 1.0 (near)
      this.depth = 0.6 + Math.random() * 0.4; 
      this.size = (20 + Math.random() * 15) * this.depth;

      if (speciesRoll < 0.33) {
        this.species = 'jelly';
        this.color = COLORS.cyan;
      } else if (speciesRoll < 0.66) {
        this.species = 'fin';
        this.color = COLORS.magenta;
      } else {
        this.species = 'shard';
        this.color = COLORS.lime;
      }

      this.angle = Math.atan2(this.vy, this.vx);
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = 0.02 + (Math.random() * 0.03);
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      this.wobble += this.wobbleSpeed;

      // Movement physics
      this.x += this.vx * this.depth;
      this.y += this.vy * this.depth;

      // Organic drift
      this.y += Math.sin(this.wobble) * 0.2 * this.depth;
      this.x += Math.cos(this.wobble * 0.5) * 0.2 * this.depth;

      // Boundary bounce
      if (this.x < 50) this.vx += 0.02;
      if (this.x > width - 50) this.vx -= 0.02;
      if (this.y < 50) this.vy += 0.02;
      if (this.y > height - 50) this.vy -= 0.02;

      // Flee from Net (Input Position)
      const dx = this.x - input.x;
      const dy = this.y - input.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      if (dist < 250 * this.depth) {
        const force = (250 - dist) / 250;
        this.vx += (dx / dist) * force * 0.5;
        this.vy += (dy / dist) * force * 0.5;
      }

      // Speed limiter
      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 3) {
        this.vx = (this.vx / speed) * 3;
        this.vy = (this.vy / speed) * 3;
      }

      this.angle = Math.atan2(this.vy, this.vx);
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.rotate(this.angle);
      
      // Depth of Field Blur
      // If depth is low (far), add blur. If high (near), sharp.
      if (this.depth < 0.8) {
        ctx.filter = `blur(${(1 - this.depth) * 8}px)`;
        ctx.globalAlpha = 0.6 + (this.depth * 0.4);
      }

      // Glow
      ctx.shadowBlur = 15 * this.depth;
      ctx.shadowColor = this.color;
      ctx.fillStyle = this.color;
      ctx.strokeStyle = 'rgba(255,255,255,0.9)';
      ctx.lineWidth = 1.5;

      const s = this.size;

      // Draw Species
      if (this.species === 'jelly') {
        ctx.beginPath();
        ctx.arc(0, -s*0.2, s, Math.PI, 0);
        ctx.lineTo(s, s*0.4);
        ctx.bezierCurveTo(s/2, s*0.8, -s/2, s*0.8, -s, s*0.4);
        ctx.fill();
        ctx.stroke();
      } 
      else if (this.species === 'fin') {
        ctx.beginPath();
        ctx.ellipse(0, 0, s*1.2, s*0.5, 0, 0, Math.PI*2);
        ctx.fill();
        ctx.beginPath();
        ctx.moveTo(-s, 0);
        ctx.lineTo(-s*1.8, -s*0.6);
        ctx.lineTo(-s*1.8, s*0.6);
        ctx.fill();
      } 
      else if (this.species === 'shard') {
        const wobble = Math.sin(this.wobble * 5) * 2;
        ctx.beginPath();
        ctx.moveTo(s, 0);
        ctx.lineTo(0, -s + wobble);
        ctx.lineTo(-s, 0);
        ctx.lineTo(0, s - wobble);
        ctx.closePath();
        ctx.fill();
        ctx.stroke();
      }

      ctx.restore();
    }
  }

  function init() {
    creatures = [];
    for(let i=0; i<CREATURE_COUNT; i++) {
      creatures.push(new Creature());
    }
    score = 0;
    scoreEl.innerText = score;
  }

  function checkCatch() {
    // Catch radius based on net size
    const catchRadius = 90; 

    creatures.forEach(c => {
      if (c.isCaught) return;
      
      const dx = c.x - input.x;
      const dy = c.y - input.y;
      const dist = Math.sqrt(dx*dx + dy*dy);

      if (dist < catchRadius) {
        catchCreature(c);
      }
    });
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    
    for(let i=0; i<25; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }

    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1000);

    setTimeout(() => {
      creature.reset();
    }, 2000);
  }

  function resetGame() {
    init();
  }

  function drawPlayerAndNet() {
    // --- 1. Draw The Net (Targeting Input Position) ---
    ctx.save();
    ctx.translate(input.x, input.y);
    
    // Net Glow
    ctx.shadowBlur = 20;
    ctx.shadowColor = input.active ? '#fff' : COLORS.cyan;
    
    // Net Shape
    ctx.strokeStyle = input.active ? '#fff' : COLORS.cyan;
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.arc(0, 0, 85, 0, Math.PI * 2);
    ctx.stroke();
    
    // Inner Mesh
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.4)';
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    ctx.moveTo(-85, 0); ctx.lineTo(85, 0);
    ctx.moveTo(0, -85); ctx.lineTo(0, 85);
    ctx.moveTo(-60, -60); ctx.lineTo(60, 60);
    ctx.moveTo(-60, 60); ctx.lineTo(60, -60);
    ctx.stroke();

    // Active State Ring
    if (input.active) {
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.5)';
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.arc(0, 0, 70, 0, Math.PI * 2);
      ctx.stroke();
    }
    ctx.restore();


    // --- 2. Draw The Connection (Net Handle) ---
    // Line from Player (Fixed) to Net (Input)
    ctx.save();
    ctx.beginPath();
    ctx.moveTo(player.x, player.y - 30); // Shoulder area
    ctx.lineTo(input.x, input.y);
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
    ctx.lineWidth = 4;
    ctx.stroke();
    // Highlight
    ctx.strokeStyle = COLORS.cyan;
    ctx.lineWidth = 1;
    ctx.stroke();
    ctx.restore();


    // --- 3. Draw The Player (Fixed Position) ---
    ctx.save();
    ctx.translate(player.x, player.y);
    
    // Player Glow
    ctx.shadowBlur = 20;
    ctx.shadowColor = COLORS.cyan;
    ctx.fillStyle = '#020408'; // Darker than bg for silhouette
    ctx.strokeStyle = COLORS.cyan;
    ctx.lineWidth = 2;

    // Body (Compact Humanoid)
    ctx.beginPath();
    // Head
    ctx.arc(0, -45, 10, 0, Math.PI*2);
    ctx.fill();
    ctx.stroke();
    
    // Torso
    ctx.moveTo(-8, -35);
    ctx.lineTo(8, -35);
    ctx.lineTo(12, -10);
    ctx.lineTo(-12, -10);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    // Legs
    ctx.beginPath();
    ctx.moveTo(-8, -10);
    ctx.lineTo(-15, 20);
    ctx.lineTo(-5, 20);
    ctx.lineTo(0, -5);
    ctx.lineTo(5, 20);
    ctx.lineTo(15, 20);
    ctx.lineTo(8, -10);
    ctx.fill();
    ctx.stroke();

    // Arm (Reaching towards net)
    // Calculate angle to net
    const angleToNet = Math.atan2(input.y - (player.y - 20), input.x - player.x);
    ctx.rotate(angleToNet);
    ctx.beginPath();
    ctx.rect(0, -3, 40, 6); // Arm length
    ctx.fill();
    ctx.stroke();

    ctx.restore();
  }

  function loop() {
    frameCount++;
    ctx.clearRect(0, 0, width, height);

    // Sort creatures by depth for correct overlap
    creatures.sort((a, b) => a.depth - b.depth);

    // Update & Draw Creatures
    creatures.forEach(c => {
      c.update();
      c.draw(ctx);
    });

    // Update & Draw Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      let p = particles[i];
      p.update();
      p.draw(ctx);
      if (p.life <= 0) particles.splice(i, 1);
    }

    // Draw Player & Net (Always on top)
    drawPlayerAndNet();

    requestAnimationFrame(loop);
  }

  // Start
  init();
  loop();

</script>
