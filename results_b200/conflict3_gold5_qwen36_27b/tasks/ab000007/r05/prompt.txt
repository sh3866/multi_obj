<script>
  /**
   * Game Logic & Rendering
   */
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  
  // Mouse/Touch state (Controls the NET position)
  let input = { x: 0, y: 0, active: false };
  
  // Player Position (Fixed lower-left)
  let player = { x: 0, y: 0 };
  let frameCount = 0;

  const COLORS = {
    cyan: '#00f0ff',
    magenta: '#ff0055',
    lime: '#ccff00',
    deepPurple: '#8800ff'
  };
  const CREATURE_COUNT = 8;

  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
    
    // Position player fixed at lower left
    player.x = width * 0.15; 
    player.y = height * 0.75;
    
    // Initialize input to player pos
    input.x = player.x + 200;
    input.y = player.y - 200;
  }
  window.addEventListener('resize', resize);
  resize();

  // Input Handling
  window.addEventListener('mousemove', e => {
    input.x = e.clientX;
    input.y = e.clientY;
  });
  window.addEventListener('mousedown', () => input.active = true);
  window.addEventListener('mouseup', () => {
    input.active = false;
    checkCatch();
  });
  
  window.addEventListener('touchmove', e => {
    e.preventDefault();
    input.x = e.touches[0].clientX;
    input.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    input.x = e.touches[0].clientX;
    input.y = e.touches[0].clientY;
    input.active = true;
  });
  window.addEventListener('touchend', () => {
    input.active = false;
    checkCatch();
  });

  // --- Classes ---

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 6 + 2;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.02 + 0.015;
      this.size = Math.random() * 3 + 1;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.vx *= 0.95;
      this.vy *= 0.95;
      this.life -= this.decay;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor() {
      this.reset(true);
    }

    reset(initial = false) {
      // Spawn logic
      if (initial) {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
      } else {
        this.x = Math.random() < 0.5 ? -100 : width + 100;
        this.y = Math.random() * (height - 100);
      }
      
      this.vx = (Math.random() - 0.5) * 1.5;
      this.vy = (Math.random() - 0.5) * 1.5;
      
      // Species & Depth
      const speciesRoll = Math.random();
      
      // Depth factor: 0.5 (far) to 1.0 (near)
      this.depth = 0.6 + Math.random() * 0.4; 
      this.size = (18 + Math.random() * 12) * this.depth;

      if (speciesRoll < 0.33) {
        this.species = 'jelly';
        this.color = COLORS.cyan;
      } else if (speciesRoll < 0.66) {
        this.species = 'fin';
        this.color = COLORS.magenta;
      } else {
        this.species = 'shard';
        this.color = COLORS.lime;
      }

      this.angle = Math.atan2(this.vy, this.vx);
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = 0.03 + (Math.random() * 0.02);
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      this.wobble += this.wobbleSpeed;

      // Movement physics
      this.x += this.vx * this.depth;
      this.y += this.vy * this.depth;

      // Organic drift
      this.y += Math.sin(this.wobble) * 0.3 * this.depth;
      this.x += Math.cos(this.wobble * 0.5) * 0.3 * this.depth;

      // Boundary bounce
      if (this.x < 50) this.vx += 0.05;
      if (this.x > width - 50) this.vx -= 0.05;
      if (this.y < 50) this.vy += 0.05;
      if (this.y > height - 50) this.vy -= 0.05;

      // Flee from Net (Input Position)
      const dx = this.x - input.x;
      const dy = this.y - input.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      if (dist < 200 * this.depth) {
        const force = (200 - dist) / 200;
        this.vx += (dx / dist) * force * 0.8;
        this.vy += (dy / dist) * force * 0.8;
      }

      // Speed limiter
      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 3.5) {
        this.vx = (this.vx / speed) * 3.5;
        this.vy = (this.vy / speed) * 3.5;
      }

      this.angle = Math.atan2(this.vy, this.vx);
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.rotate(this.angle);
      
      // Depth of Field Blur
      if (this.depth < 0.85) {
        ctx.filter = `blur(${(1 - this.depth) * 6}px)`;
        ctx.globalAlpha = 0.5 + (this.depth * 0.5);
      }

      // Glow
      ctx.shadowBlur = 12 * this.depth;
      ctx.shadowColor = this.color;
      ctx.fillStyle = this.color;
      ctx.strokeStyle = 'rgba(255,255,255,0.8)';
      ctx.lineWidth = 1.5;

      const s = this.size;

      // Draw Species
      if (this.species === 'jelly') {
        ctx.beginPath();
        ctx.arc(0, -s*0.2, s, Math.PI, 0);
        ctx.lineTo(s, s*0.4);
        ctx.bezierCurveTo(s/2, s*0.8, -s/2, s*0.8, -s, s*0.4);
        ctx.fill();
        ctx.stroke();
      } 
      else if (this.species === 'fin') {
        ctx.beginPath();
        ctx.ellipse(0, 0, s*1.2, s*0.5, 0, 0, Math.PI*2);
        ctx.fill();
        ctx.beginPath();
        ctx.moveTo(-s, 0);
        ctx.lineTo(-s*1.8, -s*0.6);
        ctx.lineTo(-s*1.8, s*0.6);
        ctx.fill();
      } 
      else if (this.species === 'shard') {
        const wobble = Math.sin(this.wobble * 5) * 2;
        ctx.beginPath();
        ctx.moveTo(s, 0);
        ctx.lineTo(0, -s + wobble);
        ctx.lineTo(-s, 0);
        ctx.lineTo(0, s - wobble);
        ctx.closePath();
        ctx.fill();
        ctx.stroke();
      }

      ctx.restore();
    }
  }

  function init() {
    creatures = [];
    for(let i=0; i<CREATURE_COUNT; i++) {
      creatures.push(new Creature());
    }
    score = 0;
    scoreEl.innerText = score;
  }

  function checkCatch() {
    // Catch radius based on net size
    const catchRadius = 80; 

    creatures.forEach(c => {
      if (c.isCaught) return;
      
      const dx = c.x - input.x;
      const dy = c.y - input.y;
      const dist = Math.sqrt(dx*dx + dy*dy);

      if (dist < catchRadius) {
        catchCreature(c);
      }
    });
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    
    for(let i=0; i<30; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }

    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1200);

    setTimeout(() => {
      creature.reset();
    }, 2500);
  }

  function resetGame() {
    init();
  }

  function drawPlayerAndNet() {
    // --- 1. Draw The Handle (Connection) ---
    // Line from Player (Fixed) to Net (Input)
    ctx.save();
    
    // Calculate angle for handle thickness
    const angle = Math.atan2(input.y - (player.y - 20), input.x - player.x);
    const perpAngle = angle + Math.PI / 2;
    const handleWidth = 4;

    // Draw thick handle line
    ctx.beginPath();
    ctx.moveTo(player.x + Math.cos(perpAngle) * handleWidth, (player.y - 20) + Math.sin(perpAngle) * handleWidth);
    ctx.lineTo(input.x + Math.cos(perpAngle) * handleWidth, input.y + Math.sin(perpAngle) * handleWidth);
    ctx.lineTo(input.x - Math.cos(perpAngle) * handleWidth, input.y - Math.sin(perpAngle) * handleWidth);
    ctx.lineTo(player.x - Math.cos(perpAngle) * handleWidth, (player.y - 20) - Math.sin(perpAngle) * handleWidth);
    ctx.closePath();
    
    // Handle body gradient
    const grad = ctx.createLinearGradient(player.x, player.y, input.x, input.y);
    grad.addColorStop(0, '#1a1a2e');
    grad.addColorStop(1, '#0f141e');
    ctx.fillStyle = grad;
    ctx.fill();
    
    // Handle edge highlight
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.3)';
    ctx.lineWidth = 1;
    ctx.stroke();
    ctx.restore();

    // --- 2. Draw The Net (Targeting Input Position) ---
    ctx.save();
    ctx.translate(input.x, input.y);
    
    // Calculate angle to player for rotation (tilt)
    // We want the net to face the player slightly or the creature
    // Let's rotate so the "back" of the net faces the player handle
    const netAngle = Math.atan2(player.y - input.y, player.x - input.x);
    ctx.rotate(netAngle);

    // Net Glow
    ctx.shadowBlur = 15;
    ctx.shadowColor = input.active ? '#fff' : COLORS.cyan;
    
    // Net Shape (Oval Mesh)
    ctx.strokeStyle = input.active ? 'rgba(255,255,255,0.8)' : COLORS.cyan;
    ctx.lineWidth = 2.5;
    ctx.beginPath();
    // Draw an ellipse: rx=60, ry=35
    ctx.ellipse(0, 0, 60, 35, 0, 0, Math.PI * 2);
    ctx.stroke();
    
    // Inner Mesh (Crosshatch)
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.25)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    
    // Draw lines that curve to simulate 3D mesh
    const rx = 60;
    const ry = 35;
    
    // Vertical-ish lines
    for(let i = -40; i <= 40; i+=10) {
        ctx.moveTo(i, -Math.sqrt(1 - (i/rx)**2)*ry);
        ctx.lineTo(i, Math.sqrt(1 - (i/rx)**2)*ry);
    }
    // Horizontal-ish lines
    for(let i = -20; i <= 20; i+=10) {
        ctx.moveTo(-Math.sqrt(1 - (i/ry)**2)*rx, i);
        ctx.lineTo(Math.sqrt(1 - (i/ry)**2)*rx, i);
    }
    ctx.stroke();

    // Active State (Snap effect)
    if (input.active) {
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.8)';
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.ellipse(0, 0, 40, 25, 0, 0, Math.PI * 2);
      ctx.stroke();
      
      // Inner flash
      ctx.fillStyle = 'rgba(255, 255, 255, 0.1)';
      ctx.fill();
    }
    ctx.restore();

    // --- 3. Draw The Player (Fixed Position) ---
    ctx.save();
    ctx.translate(player.x, player.y);
    
    // Player Glow
    ctx.shadowBlur = 20;
    ctx.shadowColor = COLORS.cyan;
    ctx.fillStyle = '#020408'; // Darker than bg for silhouette
    ctx.strokeStyle = COLORS.cyan;
    ctx.lineWidth = 2;

    // Body (Compact Humanoid)
    ctx.beginPath();
    // Head
    ctx.arc(0, -50, 12, 0, Math.PI*2);
    ctx.fill();
    ctx.stroke();
    
    // Torso
    ctx.moveTo(-10, -38);
    ctx.lineTo(10, -38);
    ctx.lineTo(14, -15);
    ctx.lineTo(-14, -15);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    // Legs
    ctx.beginPath();
    ctx.moveTo(-10, -15);
    ctx.lineTo(-18, 25);
    ctx.lineTo(-8, 25);
    ctx.lineTo(0, 5);
    ctx.lineTo(8, 25);
    ctx.lineTo(18, 25);
    ctx.lineTo(10, -15);
    ctx.fill();
    ctx.stroke();

    // Arm (Reaching towards net)
    // We draw the arm last so it overlaps the body
    const angleToNet = Math.atan2(input.y - (player.y - 20), input.x - player.x);
    
    ctx.save();
    ctx.translate(0, -30); // Shoulder pivot
    ctx.rotate(angleToNet);
    
    // Draw Arm segment
    ctx.beginPath();
    ctx.rect(0, -4, 35, 8); 
    ctx.fill();
    ctx.stroke();
    
    // Draw Hand (simple oval)
    ctx.beginPath();
    ctx.ellipse(38, 0, 6, 4, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.stroke();
    
    ctx.restore(); // Restore arm context

    ctx.restore(); // Restore player context
  }

  function loop() {
    frameCount++;
    ctx.clearRect(0, 0, width, height);

    // Sort creatures by depth for correct overlap
    creatures.sort((a, b) => a.depth - b.depth);

    // Update & Draw Creatures
    creatures.forEach(c => {
      c.update();
      c.draw(ctx);
    });

    // Update & Draw Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      let p = particles[i];
      p.update();
      p.draw(ctx);
      if (p.life <= 0) particles.splice(i, 1);
    }

    // Draw Player & Net (Always on top)
    drawPlayerAndNet();

    requestAnimationFrame(loop);
  }

  // Start
  init();
  loop();
