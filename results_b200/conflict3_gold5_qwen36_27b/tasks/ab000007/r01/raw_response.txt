<script>
  /**
   * Game Logic & Rendering
   * Pure Canvas implementation.
   */
  
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  let mouse = { x: window.innerWidth / 2, y: window.innerHeight - 100 };
  let isMouseDown = false;
  let frameCount = 0;

  // Configuration
  const COLORS = {
    cyan: '#00f0ff',
    magenta: '#ff0055',
    lime: '#ccff00',
    deepPurple: '#8800ff'
  };
  const CREATURE_COUNT = 7;

  // Resize Handling
  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
  }
  window.addEventListener('resize', resize);
  resize();

  // Input Handling
  window.addEventListener('mousemove', e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
  });
  window.addEventListener('mousedown', () => isMouseDown = true);
  window.addEventListener('mouseup', () => {
    isMouseDown = false;
    checkCatch();
  });
  
  // Touch support
  window.addEventListener('touchmove', e => {
    e.preventDefault();
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
    isMouseDown = true;
  });
  window.addEventListener('touchend', () => {
    isMouseDown = false;
    checkCatch();
  });

  // --- Classes ---

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 4 + 2;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.03 + 0.02;
      this.size = Math.random() * 3 + 1;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.vx *= 0.95;
      this.vy *= 0.95;
      this.life -= this.decay;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor() {
      this.reset(true);
    }

    reset(initial = false) {
      // Spawn logic
      if (initial) {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
      } else {
        this.x = Math.random() < 0.5 ? -100 : width + 100;
        this.y = Math.random() * (height - 150); // Keep away from very bottom
      }
      
      this.vx = (Math.random() - 0.5) * 1.5;
      this.vy = (Math.random() - 0.5) * 1.5;
      
      // Species definition
      const speciesRoll = Math.random();
      if (speciesRoll < 0.33) {
        this.species = 'jelly';
        this.color = COLORS.cyan;
        this.size = 30;
      } else if (speciesRoll < 0.66) {
        this.species = 'fin';
        this.color = COLORS.magenta;
        this.size = 25;
      } else {
        this.species = 'shard';
        this.color = COLORS.lime;
        this.size = 20;
      }

      this.angle = Math.atan2(this.vy, this.vx);
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = 0.03 + Math.random() * 0.02;
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      this.wobble += this.wobbleSpeed;

      // Movement physics
      this.x += this.vx;
      this.y += this.vy;

      // Organic drift
      this.y += Math.sin(this.wobble) * 0.3;
      this.x += Math.cos(this.wobble * 0.5) * 0.3;

      // Boundary bounce (soft)
      if (this.x < 50) this.vx += 0.05;
      if (this.x > width - 50) this.vx -= 0.05;
      if (this.y < 50) this.vy += 0.05;
      if (this.y > height - 100) this.vy -= 0.05; // Avoid player area slightly

      // Flee from player net
      const dx = this.x - mouse.x;
      const dy = this.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      if (dist < 200) {
        const force = (200 - dist) / 200;
        // Repulsion
        this.vx += (dx / dist) * force * 0.8;
        this.vy += (dy / dist) * force * 0.8;
      }

      // Speed limiter
      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 3.5) {
        this.vx = (this.vx / speed) * 3.5;
        this.vy = (this.vy / speed) * 3.5;
      }

      // Update facing angle
      this.angle = Math.atan2(this.vy, this.vx);
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.rotate(this.angle);
      
      // Glow effect
      ctx.shadowBlur = 20;
      ctx.shadowColor = this.color;
      ctx.fillStyle = this.color;
      ctx.strokeStyle = 'rgba(255,255,255,0.8)';
      ctx.lineWidth = 1.5;

      // Draw Species Silhouette
      if (this.species === 'jelly') {
        // Bell shape
        ctx.beginPath();
        ctx.arc(0, -5, this.size, Math.PI, 0);
        ctx.lineTo(this.size, 5);
        ctx.bezierCurveTo(this.size/2, 15, -this.size/2, 15, -this.size, 5);
        ctx.fill();
        ctx.stroke();
        // Inner detail
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(-5, -5, 4, 0, Math.PI*2);
        ctx.fill();
      } 
      else if (this.species === 'fin') {
        // Fish/Ray shape
        ctx.beginPath();
        ctx.ellipse(0, 0, this.size * 1.2, this.size * 0.6, 0, 0, Math.PI*2);
        ctx.fill();
        // Tail
        ctx.beginPath();
        ctx.moveTo(-this.size, 0);
        ctx.lineTo(-this.size * 1.8, -10);
        ctx.lineTo(-this.size * 1.8, 10);
        ctx.fill();
      } 
      else if (this.species === 'shard') {
        // Geometric floating crystal
        const s = this.size;
        const wobble = Math.sin(this.wobble * 5) * 2;
        ctx.beginPath();
        ctx.moveTo(s, 0);
        ctx.lineTo(0, -s + wobble);
        ctx.lineTo(-s, 0);
        ctx.lineTo(0, s - wobble);
        ctx.closePath();
        ctx.fill();
        ctx.stroke();
        // Core
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(0, 0, s/3, 0, Math.PI*2);
        ctx.fill();
      }

      ctx.restore();
    }
  }

  // --- Game Functions ---

  function init() {
    creatures = [];
    for(let i=0; i<CREATURE_COUNT; i++) {
      creatures.push(new Creature());
    }
    score = 0;
    scoreEl.innerText = score;
  }

  function checkCatch() {
    const catchRadius = 80; // Net capture radius

    creatures.forEach(c => {
      if (c.isCaught) return;
      
      const dx = c.x - mouse.x;
      const dy = c.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);

      if (dist < catchRadius) {
        catchCreature(c);
      }
    });
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    
    // Visual FX
    for(let i=0; i<20; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }

    // Show Toast
    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1200);

    // Respawn logic
    setTimeout(() => {
      creature.reset();
    }, 2500);
  }

  function resetGame() {
    init();
  }

  function drawPlayerAndNet() {
    // 1. Draw Player Silhouette (Lower Field)
    // We draw this statically relative to the mouse Y, or fixed at bottom?
    // Brief says "player with a net". Let's draw the player silhouette near the mouse position
    // but slightly below the net center to imply they are holding it.
    
    ctx.save();
    ctx.translate(mouse.x, mouse.y + 30); // Offset from center of net
    
    // Silhouette
    ctx.fillStyle = 'rgba(0, 0, 0, 0.8)';
    ctx.shadowBlur = 10;
    ctx.shadowColor = 'rgba(0, 240, 255, 0.5)';
    
    // Simple hooded figure shape
    ctx.beginPath();
    ctx.arc(0, -20, 12, 0, Math.PI*2); // Head
    ctx.fill();
    ctx.beginPath();
    ctx.moveTo(-15, -10);
    ctx.quadraticCurveTo(0, 10, 15, -10); // Shoulders
    ctx.lineTo(20, 25); // Arm right
    ctx.lineTo(-20, 25); // Arm left
    ctx.fill();

    ctx.restore();

    // 2. Draw The Net
    ctx.save();
    ctx.translate(mouse.x, mouse.y);
    
    // Net Glow
    ctx.shadowBlur = 15;
    ctx.shadowColor = isMouseDown ? '#fff' : COLORS.cyan;
    
    // Net Shape (Circle with grid)
    ctx.strokeStyle = isMouseDown ? '#fff' : COLORS.cyan;
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.arc(0, 0, 80, 0, Math.PI * 2);
    ctx.stroke();
    
    // Inner Grid (The Mesh)
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.3)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    // Cross lines
    ctx.moveTo(-80, 0); ctx.lineTo(80, 0);
    ctx.moveTo(0, -80); ctx.lineTo(0, 80);
    // Diagonals
    ctx.moveTo(-56, -56); ctx.lineTo(56, 56);
    ctx.moveTo(-56, 56); ctx.lineTo(56, -56);
    ctx.stroke();

    // "Sucking" effect if held down
    if (isMouseDown) {
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.4)';
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.arc(0, 0, 60, 0, Math.PI * 2);
      ctx.stroke();
    }

    ctx.restore();
  }

  function loop() {
    frameCount++;
    // Clear
    ctx.clearRect(0, 0, width, height);

    // Update & Draw Creatures
    creatures.forEach(c => {
      c.update();
      c.draw(ctx);
    });

    // Update & Draw Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      let p = particles[i];
      p.update();
      p.draw(ctx);
      if (p.life <= 0) particles.splice(i, 1);
    }

    // Draw Player & Net
    drawPlayerAndNet();

    requestAnimationFrame(loop);
  }

  // Start
  init();
  loop();

</script>
