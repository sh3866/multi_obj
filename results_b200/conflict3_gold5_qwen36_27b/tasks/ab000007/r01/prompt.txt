<script>
  /**
   * Game Logic & Rendering
   * Pure Canvas implementation for smooth 60fps rendering.
   */
  
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  let mouse = { x: -1000, y: -1000 };
  let isMouseDown = false;

  // Configuration
  const COLORS = ['#00f0ff', '#ff0055', '#ccff00', '#bd00ff'];
  const CREATURE_COUNT = 6;

  // Resize Handling
  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
  }
  window.addEventListener('resize', resize);
  resize();

  // Input Handling
  window.addEventListener('mousemove', e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
  });
  window.addEventListener('mousedown', () => isMouseDown = true);
  window.addEventListener('mouseup', () => {
    isMouseDown = false;
    checkCatch();
  });
  // Touch support
  window.addEventListener('touchmove', e => {
    e.preventDefault(); // Prevent scrolling
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
    isMouseDown = true;
  });
  window.addEventListener('touchend', () => {
    isMouseDown = false;
    checkCatch();
  });

  // --- Classes ---

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 3 + 1;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.03 + 0.01;
      this.size = Math.random() * 3 + 1;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.life -= this.decay;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor() {
      this.reset();
      // Start at random positions initially
      this.x = Math.random() * width;
      this.y = Math.random() * height;
    }

    reset() {
      // Spawn away from center slightly
      this.x = Math.random() < 0.5 ? -50 : width + 50;
      this.y = Math.random() * height;
      this.vx = (Math.random() - 0.5) * 2;
      this.vy = (Math.random() - 0.5) * 2;
      this.radius = Math.random() * 20 + 15;
      this.color = COLORS[Math.floor(Math.random() * COLORS.length)];
      this.angle = 0;
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = Math.random() * 0.05 + 0.02;
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      // Movement
      this.x += this.vx;
      this.y += this.vy;
      this.wobble += this.wobbleSpeed;
      
      // Sine wave movement for organic feel
      this.x += Math.sin(this.wobble) * 0.5;

      // Bounce off walls
      if (this.x < 0 || this.x > width) this.vx *= -1;
      if (this.y < 0 || this.y > height) this.vy *= -1;

      // Avoid Mouse (Net)
      const dx = this.x - mouse.x;
      const dy = this.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      if (dist < 150) {
        // Flee behavior
        const force = (150 - dist) / 150;
        this.vx += (dx / dist) * force * 0.5;
        this.vy += (dy / dist) * force * 0.5;
      }

      // Speed limiter
      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 4) {
        this.vx = (this.vx / speed) * 4;
        this.vy = (this.vy / speed) * 4;
      }
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      
      // Glow
      ctx.shadowBlur = 20;
      ctx.shadowColor = this.color;
      
      // Body
      ctx.fillStyle = this.color;
      ctx.beginPath();
      
      // Draw a slightly organic blob shape
      const r = this.radius;
      const wobbleOffset = Math.sin(this.wobble * 2) * 5;
      
      ctx.moveTo(r + wobbleOffset, 0);
      ctx.bezierCurveTo(r, r, 0, r, -r - wobbleOffset, 0);
      ctx.bezierCurveTo(-r, -r, 0, -r, r + wobbleOffset, 0);
      
      ctx.fill();

      // Inner Core
      ctx.fillStyle = '#fff';
      ctx.shadowBlur = 0;
      ctx.beginPath();
      ctx.arc(0, 0, r * 0.3, 0, Math.PI * 2);
      ctx.fill();

      ctx.restore();
    }
  }

  // --- Game Functions ---

  function init() {
    creatures = [];
    for(let i=0; i<CREATURE_COUNT; i++) {
      creatures.push(new Creature());
    }
    score = 0;
    scoreEl.innerText = score;
  }

  function checkCatch() {
    // Simple collision detection with the "Net" radius
    const catchRadius = 60; // Size of the net capture zone

    creatures.forEach(c => {
      if (c.isCaught) return;
      
      const dx = c.x - mouse.x;
      const dy = c.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);

      if (dist < catchRadius) {
        catchCreature(c);
      }
    });
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    
    // Visual FX
    for(let i=0; i<15; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }

    // Show Toast
    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1000);

    // Respawn logic (delayed)
    setTimeout(() => {
      creature.reset();
    }, 2000);
  }

  function resetGame() {
    init();
  }

  function drawNet() {
    // Draw the player's net/cursor
    ctx.save();
    ctx.translate(mouse.x, mouse.y);
    
    // Net Glow
    ctx.shadowBlur = 15;
    ctx.shadowColor = '#00f0ff';
    
    // Net Shape (Circle with dashed rim)
    ctx.strokeStyle = isMouseDown ? '#fff' : '#00f0ff';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.arc(0, 0, 60, 0, Math.PI * 2);
    ctx.stroke();
    
    // Inner reticle
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.3)';
    ctx.beginPath();
    ctx.arc(0, 0, 40, 0, Math.PI * 2);
    ctx.stroke();

    // Crosshair
    ctx.beginPath();
    ctx.moveTo(-10, 0); ctx.lineTo(10, 0);
    ctx.moveTo(0, -10); ctx.lineTo(0, 10);
    ctx.stroke();

    // "Sucking" effect if held down
    if (isMouseDown) {
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.1)';
      ctx.beginPath();
      ctx.arc(0, 0, 80, 0, Math.PI * 2);
      ctx.stroke();
    }

    ctx.restore();
  }

  function loop() {
    // Clear
    ctx.clearRect(0, 0, width, height);

    // Update & Draw Creatures
    creatures.forEach(c => {
      c.update();
      c.draw(ctx);
    });

    // Update & Draw Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      let p = particles[i];
      p.update();
      p.draw(ctx);
      if (p.life <= 0) particles.splice(i, 1);
    }

    // Draw Player Net
    drawNet();

    requestAnimationFrame(loop);
  }

  // Start
  init();
  loop();

</script>
