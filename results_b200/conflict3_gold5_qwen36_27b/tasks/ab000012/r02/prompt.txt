<script>
    // --- Background Particle System (Dust & Magic) ---
    const canvas = document.getElementById('canvas-bg');
    const ctx = canvas.getContext('2d');

    let width, height;
    let particles = [];

    function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;
    }

    class Particle {
        constructor() {
            this.reset();
        }

        reset() {
            this.x = Math.random() * width;
            this.y = Math.random() * height;
            this.vx = (Math.random() - 0.5) * 0.2; // Slower drift
            this.vy = (Math.random() - 0.5) * 0.2;
            this.size = Math.random() * 1.5;
            this.alpha = Math.random() * 0.4;
            this.life = Math.random() * 100 + 50;
            // Mix of Gold and Teal particles
            const isGold = Math.random() > 0.7;
            this.color = isGold ? '212, 175, 55' : '79, 209, 197';
        }

        update() {
            this.x += this.vx;
            this.y += this.vy;
            this.life--;

            // Subtle fade in/out
            if (this.life < 20) {
                this.alpha -= 0.02;
            }

            if (this.life <= 0 || this.alpha <= 0) {
                this.reset();
            }
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(${this.color}, ${this.alpha})`;
            ctx.fill();
        }
    }

    function initParticles() {
        particles = [];
        const count = Math.floor((width * height) / 12000); // Slightly higher density
        for (let i = 0; i < count; i++) {
            particles.push(new Particle());
        }
    }

    function animate() {
        ctx.clearRect(0, 0, width, height);
        
        // Draw subtle connecting lines for "magic" feel (only if close)
        ctx.lineWidth = 0.5;
        for (let i = 0; i < particles.length; i++) {
            particles[i].update();
            particles[i].draw();
            
            // Optimization: only check connections for a subset or close neighbors to save perf
            if (i % 5 === 0) { 
                for (let j = i + 1; j < particles.length; j+=5) {
                    const dx = particles[i].x - particles[j].x;
                    const dy = particles[i].y - particles[j].y;
                    // Fast distance check
                    if (dx*dx + dy*dy < 2500) { // 50px radius squared
                        ctx.beginPath();
                        ctx.strokeStyle = `rgba(79, 209, 197, ${0.05})`;
                        ctx.moveTo(particles[i].x, particles[i].y);
                        ctx.lineTo(particles[j].x, particles[j].y);
                        ctx.stroke();
                    }
                }
            }
        }
        requestAnimationFrame(animate);
    }

    // --- Interactions ---
    window.addEventListener('resize', () => {
        resize();
        initParticles();
    });

    function interact(action) {
        // Visual feedback for buttons
        const btn = event.currentTarget;
        btn.style.transform = 'scale(0.96)';
        
        // Flash effect
        const originalBorder = btn.style.borderColor;
        btn.style.borderColor = 'var(--gold-highlight)';
        
        setTimeout(() => {
            btn.style.transform = '';
            btn.style.borderColor = originalBorder;
        }, 200);
        
        console.log(`Action triggered: ${action}`);
    }

    // Initialize
    resize();
    initParticles();
    animate();

</script>
