 * Spherical Conway's Game of Life - Revised Design
 * 
 * Design Improvements:
 * 1. Increased Cyan/Amber presence via biased initialization.
 * 2. Asymmetric lighting (Top-Left) to create spherical volume.
 * 3. Rim illumination (Fresnel-like effect) for depth.
 * 4. Subtle rear-surface dimming.
 * 5. Cleaner, tighter UI.
 */

// --- Configuration ---
const CONFIG = {
  cellCount: 3800, // Slightly higher density
  sphereRadius: 260,
  cellSize: 3.0,
  rotationSpeed: 0.0005,
  colors: {
    cyan: { h: 185, s: 100, l: 65 },   // New
    violet: { h: 270, s: 90, l: 70 },  // Stable
    magenta: { h: 330, s: 100, l: 65 },// Aging
    amber: { h: 35, s: 100, l: 60 },   // Dying
  }
};

// --- State ---
let state = {
  cells: [], 
  generation: 0,
  isRunning: true,
  speed: 30, 
  lastFrameTime: 0,
  rotation: { x: 0.6, y: 0.8 }, // Initial tilt
  zoom: 1,
  isDragging: false,
  lastMouse: { x: 0, y: 0 }
};

// --- Canvas Setup ---

const ctx = canvas.getContext('2d', { alpha: false });

// --- Math Helpers ---
const Vec3 = {
  distSq: (v1, v2) => {
    const dx = v1.x - v2.x;
    const dy = v1.y - v2.y;
    const dz = v1.z - v2.z;
    return dx*dx + dy*dy + dz*dz;
  }
};

// Fibonacci Lattice
function createSpherePoints(count, radius) {
  const points = [];
  const goldenAngle = Math.PI * (3 - Math.sqrt(5));
  
  for (let i = 0; i < count; i++) {
    const y = 1 - (i / (count - 1)) * 2;
    const radiusAtY = Math.sqrt(1 - y * y);
    const theta = goldenAngle * i;
    
    points.push({
      x: Math.cos(theta) * radiusAtY * radius,
      y: y * radius,
      z: Math.sin(theta) * radiusAtY * radius
    });
  }
  return points;
}

// --- Game Logic ---

function initGame() {
  const points = createSpherePoints(CONFIG.cellCount, CONFIG.sphereRadius);
  
  // Calculate neighbor threshold
  const thresholdSq = Math.pow(22, 2);

  state.cells = points.map((p, i) => {
    const neighbors = [];
    for (let j = 0; j < points.length; j++) {
      if (i === j) continue;
      if (Vec3.distSq(p, points[j]) < thresholdSq) { 
        neighbors.push(j);
      }
    }
    
    // --- DESIGN REVISED INITIALIZATION ---
    // To ensure Cyan and Amber are visible and not just token clusters,
    // we use a biased noise pattern.
    
    let isAlive = Math.random() > 0.4; // 60% density
    let age = 0;
    let deathTimer = 0;

    // Force specific states based on position to create "regions"
    // This prevents the simulation from looking purely random and ensures color variety
    const yNorm = p.y / CONFIG.sphereRadius; // -1 to 1
    const theta = Math.atan2(p.z, p.x);
    
    // Create "Storms" of new cells (Cyan) in specific latitudes
    if (Math.abs(yNorm) < 0.2 && Math.random() > 0.3) {
      isAlive = true;
      age = 1; // Cyan
    }
    // Create "Dying" regions (Amber) at the poles
    else if (Math.abs(yNorm) > 0.8 && Math.random() > 0.5) {
      isAlive = false;
      deathTimer = 4;
    }
    // Random mix elsewhere
    else if (isAlive) {
      age = Math.floor(Math.random() * 20);
    }

    return {
      pos: p,
      neighbors: neighbors,
      alive: isAlive,
      age: age, 
      deathTimer: deathTimer
    };
  });
  
  state.generation = 0;
  updateStats();
}

function stepGame() {
  const updates = [];
  
  state.cells.forEach((cell, i) => {
    const neighborCount = cell.neighbors.reduce((acc, idx) => acc + (state.cells[idx].alive ? 1 : 0), 0);
    
    let nextAlive = false;
    let isBorn = false;

    // Standard Conway Rules
    if (cell.alive) {
      if (neighborCount < 2 || neighborCount > 3) {
        nextAlive = false;
      } else {
        nextAlive = true;
      }
    } else {
      if (neighborCount === 3) {
        nextAlive = true;
        isBorn = true;
      }
    }

    updates.push({
      alive: nextAlive,
      isBorn: isBorn,
      dying: nextAlive === false && cell.alive === true
    });
  });

  // Apply updates
  state.cells.forEach((cell, i) => {
    const u = updates[i];
    
    if (u.isBorn) {
      cell.alive = true;
      cell.age = 0; // Cyan
      cell.deathTimer = 0;
    } else if (u.dying) {
      cell.alive = false;
      cell.deathTimer = 4; // Amber duration
      cell.age = 0;
    } else if (cell.alive) {
      cell.age++;
    } else if (cell.deathTimer > 0) {
      cell.deathTimer--;
    }
  });

  state.generation++;
  updateStats();
}

// --- Rendering ---

function project(p, rotX, rotY, zoom) {
  // Rotate Y
  let x = p.x * Math.cos(rotY) - p.z * Math.sin(rotY);
  let z = p.z * Math.cos(rotY) + p.x * Math.sin(rotY);
  
  // Rotate X
  let y = p.y * Math.cos(rotX) - z * Math.sin(rotX);
  let zFinal = z * Math.cos(rotX) + p.y * Math.sin(rotX);
  
  // Perspective
  const fov = 1000 * zoom;
  const scale = fov / (fov + zFinal);
  
  return {
    x: x * scale + width / 2,
    y: y * scale + height / 2,
    scale: scale,
    z: zFinal 
  };
}

function draw() {
  // Clear with a slight trail for motion blur on the background
  ctx.fillStyle = 'rgba(4, 4, 6, 0.5)'; 
  ctx.fillRect(0, 0, width, height);

  // Prepare projected cells
  const projectedCells = state.cells.map(cell => {
    const proj = project(cell.pos, state.rotation.x, state.rotation.y, state.zoom);
    return { cell, proj };
  });
  
  // Sort by Z (Painter's Algorithm)
  projectedCells.sort((a, b) => b.proj.z - a.proj.z);

  projectedCells.forEach(({ cell, proj }) => {
    // Cull backface completely if too far back, but let near-back show
    if (proj.z > CONFIG.sphereRadius * 0.9) return; 

    const size = CONFIG.cellSize * proj.scale;
    
    // --- REVISED DEPTH & LIGHTING MODEL ---
    
    // 1. Depth Ratio (0 = front, 1 = back)
    const depthRatio = (proj.z + CONFIG.sphereRadius) / (2 * CONFIG.sphereRadius);
    
    // 2. Asymmetric Lighting (Top-Left)
    // We calculate a normalized vector from the cell position to the light source
    // Light source is roughly at (-1, -1, -2) in world space relative to sphere center
    // We approximate by checking x and y coordinates relative to center
    // Front-left is brighter.
    const lightFactor = 1 - ( (proj.x - width/2) * 0.0005 + (proj.y - height/2) * 0.0005 ); 
    
    // 3. Alpha Falloff
    // Front is solid, back fades out
    const alpha = Math.max(0.05, 1 - Math.pow(depthRatio, 1.2));
    
    // 4. Brightness Falloff (Terminator line)
    // Cells on the far edge should be darker
    const brightness = Math.max(0.3, lightFactor * (1 - depthRatio * 0.5));
    
    let colorStr;
    let shadowColor = "transparent";
    let shadowBlur = 0;

    if (cell.alive) {
      let cBase;
      
      // Determine State Color
      if (cell.age < 3) {
        cBase = CONFIG.colors.cyan; 
      } else if (cell.age < 15) {
        cBase = CONFIG.colors.violet; 
      } else {
        cBase = CONFIG.colors.magenta; 
      }
      
      // Apply lighting to color
      const h = cBase.h;
      const s = cBase.s;
      const l = Math.min(95, Math.max(10, cBase.l * brightness)); 
      
      colorStr = `hsla(${h}, ${s}%, ${l}%, ${alpha})`;
      
      // Glow effect for front, bright cells
      if (depthRatio < 0.5 && brightness > 0.7) {
        shadowColor = `hsla(${h}, ${s}%, 80%, ${alpha * 0.6})`;
        shadowBlur = 10 * proj.scale;
      }
    } else {
      // Dying State (Amber)
      if (cell.deathTimer > 0) {
        const cBase = CONFIG.colors.amber;
        // Dying cells are bright regardless of lighting (bioluminescence effect)
        const l = Math.min(90, cBase.l * (brightness * 1.2)); 
        colorStr = `hsla(${cBase.h}, ${cBase.s}%, ${l}%, ${alpha})`;
        shadowColor = `hsla(${cBase.h}, ${cBase.s}%, 80%, ${alpha * 0.5})`;
        shadowBlur = 8 * proj.scale;
      } else {
        // Dead - subtle grid points
        // Dimmer in the back
        const deadAlpha = alpha * 0.15;
        const deadLight = 20 * brightness;
        colorStr = `rgba(${deadLight}, ${deadLight+2}, ${deadLight+5}, ${deadAlpha})`;
      }
    }
    
    ctx.fillStyle = colorStr;
    ctx.shadowColor = shadowColor;
    ctx.shadowBlur = shadowBlur;
    
    ctx.beginPath();
    ctx.arc(proj.x, proj.y, size, 0, Math.PI * 2);
    ctx.fill();
  });
  
  ctx.shadowBlur = 0;
  
  // --- Atmospheric Sphere Shell ---
  const cx = width / 2;
  const cy = height / 2;
  const r = CONFIG.sphereRadius * state.zoom;
  
  // 1. Outer Glow / Rim Light (Fresnel effect)
  // Cyan-tinted rim to emphasize the spherical boundary
  const rimGrad = ctx.createRadialGradient(cx, cy, r * 0.85, cx, cy, r * 1.05);
  rimGrad.addColorStop(0, 'rgba(0,0,0,0)');
  rimGrad.addColorStop(0.8, 'rgba(0, 240, 255, 0.05)'); // Subtle cyan rim
  rimGrad.addColorStop(1, 'rgba(0,0,0,0)');
  
  ctx.fillStyle = rimGrad;
  ctx.beginPath();
  ctx.arc(cx, cy, r * 1.05, 0, Math.PI * 2);
  ctx.fill();

  // 2. Inner Volume Shadow (creates the 3D sphere feel)
  // This gradient is offset to match the lighting model (Top-Left light -> Bottom-Right shadow)
  const innerGrad = ctx.createRadialGradient(cx - r*0.4, cy - r*0.4, r*0.1, cx, cy, r);
  innerGrad.addColorStop(0, 'rgba(0,0,0,0)'); // Light source side is transparent
  innerGrad.addColorStop(0.6, 'rgba(0,0,0,0.1)');
  innerGrad.addColorStop(1, 'rgba(0,0,0,0.6)'); // Far side is dark
  
  ctx.fillStyle = innerGrad;
  ctx.beginPath();
  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  ctx.fill();
}

// --- Main Loop ---

function loop(timestamp) {
  if (state.isRunning) {
    // Auto rotation
    state.rotation.y += CONFIG.rotationSpeed;
    
    // Update Logic based on speed
    if (timestamp - state.lastFrameTime > (1000 / state.speed)) {
      stepGame();
      state.lastFrameTime = timestamp;
    }
  }
  
  draw();
  requestAnimationFrame(loop);
}

}

  if (!state.isDragging) return;
  const deltaX = e.clientX - state.lastMouse.x;
  const deltaY = e.clientY - state.lastMouse.y;
  
  state.rotation.y += deltaX * 0.005;
  state.rotation.x += deltaY * 0.005;
  
  state.lastMouse = { x: e.clientX, y: e.clientY };
});

window.addEventListener('mouseup', () => {
  state.isDragging = false;
});

// Zoom
container.addEventListener('wheel', (e) => {
  e.preventDefault();
  const zoomSensitivity = 0.001;
  state.zoom -= e.deltaY * zoomSensitivity;
  state.zoom = Math.max(0.5, Math.min(3, state.zoom));
});

initGame();
requestAnimationFrame(loop);
