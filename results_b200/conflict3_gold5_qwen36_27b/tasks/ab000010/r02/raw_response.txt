<script>
/**
 * Spherical Conway's Game of Life - Design Revised
 * 
 * Visual Features:
 * - 4-State Color System (Cyan/Violet/Magenta/Amber)
 * - Depth-based shading (Z-fog)
 * - Atmospheric rim lighting
 * - Smooth trails
 */

// --- Configuration ---
const CONFIG = {
  cellCount: 4000, // Higher density for better visuals
  sphereRadius: 260,
  cellSize: 2.8,
  rotationSpeed: 0.0008,
  colors: {
    cyan: { h: 185, s: 100, l: 60 },   // New
    violet: { h: 275, s: 85, l: 65 },  // Stable
    magenta: { h: 310, s: 100, l: 65 },// Aging
    amber: { h: 35, s: 100, l: 60 },   // Dying
    dead: 'rgba(30, 30, 40, 0.3)'      // Subtle grid
  }
};

// --- State ---
let state = {
  cells: [], 
  generation: 0,
  isRunning: true,
  speed: 25, 
  lastFrameTime: 0,
  rotation: { x: 0, y: 0 },
  zoom: 1,
  isDragging: false,
  lastMouse: { x: 0, y: 0 }
};

// --- Canvas Setup ---
const canvas = document.getElementById('sphereCanvas');
const ctx = canvas.getContext('2d', { alpha: false });

let width, height;

function resize() {
  width = window.innerWidth;
  height = window.innerHeight;
  canvas.width = width;
  canvas.height = height;
}
window.addEventListener('resize', resize);
resize();

// --- Math Helpers ---
const Vec3 = {
  add: (v1, v2) => ({ x: v1.x + v2.x, y: v1.y + v2.y, z: v1.z + v2.z }),
  sub: (v1, v2) => ({ x: v1.x - v2.x, y: v1.y - v2.y, z: v1.z - v2.z }),
  dot: (v1, v2) => v1.x * v2.x + v1.y * v2.y + v1.z * v2.z,
  len: (v) => Math.sqrt(v.x*v.x + v.y*v.y + v.z*v.z),
  dist: (v1, v2) => {
    const dx = v1.x - v2.x;
    const dy = v1.y - v2.y;
    const dz = v1.z - v2.z;
    return Math.sqrt(dx*dx + dy*dy + dz*dz);
  }
};

// Fibonacci Lattice
function createSpherePoints(count, radius) {
  const points = [];
  const goldenAngle = Math.PI * (3 - Math.sqrt(5));
  
  for (let i = 0; i < count; i++) {
    const y = 1 - (i / (count - 1)) * 2;
    const radiusAtY = Math.sqrt(1 - y * y);
    const theta = goldenAngle * i;
    
    points.push({
      x: Math.cos(theta) * radiusAtY * radius,
      y: y * radius,
      z: Math.sin(theta) * radiusAtY * radius
    });
  }
  return points;
}

// --- Game Logic ---

function initGame() {
  const points = createSpherePoints(CONFIG.cellCount, CONFIG.sphereRadius);
  
  // Determine neighbor threshold based on density
  // For N points on sphere area 4*pi*R^2, average area per point is A/N. 
  // Approx distance is sqrt(Area).
  const avgDist = Math.sqrt((4 * Math.PI * Math.pow(CONFIG.sphereRadius, 2)) / CONFIG.cellCount);
  const threshold = avgDist * 1.15; // Slightly larger to ensure connectivity

  state.cells = points.map((p, i) => {
    const neighbors = [];
    // Brute force neighbor search
    for (let j = 0; j < points.length; j++) {
      if (i === j) continue;
      // Optimization: early exit if Y diff is too large? No, sphere wraps.
      // Just check distance.
      const d = Vec3.dist(p, points[j]);
      if (d < threshold) { 
        neighbors.push(j);
      }
    }
    
    return {
      pos: p,
      neighbors: neighbors,
      alive: Math.random() > 0.6, // 40% density
      age: 0, // How long it has been alive
      isDying: false // Flag to trigger amber color
    };
  });
  
  state.generation = 0;
  updateStats();
}

function stepGame() {
  // Calculate next state
  const updates = [];
  
  state.cells.forEach((cell, i) => {
    const neighborCount = cell.neighbors.reduce((acc, idx) => acc + (state.cells[idx].alive ? 1 : 0), 0);
    
    let nextAlive = false;
    let isBorn = false;

    // Standard Conway Rules
    if (cell.alive) {
      if (neighborCount < 2 || neighborCount > 3) {
        nextAlive = false;
        cell.isDying = true; // Mark for amber color one last frame
      } else {
        nextAlive = true;
        cell.isDying = false;
      }
    } else {
      if (neighborCount === 3) {
        nextAlive = true;
        isBorn = true;
      }
    }

    updates.push({
      alive: nextAlive,
      isBorn: isBorn,
      isDying: !nextAlive && cell.alive // It was alive, now dead -> Dying state
    });
  });

  // Apply updates
  state.cells.forEach((cell, i) => {
    const u = updates[i];
    const wasAlive = cell.alive;
    cell.alive = u.alive;
    
    if (u.isBorn) {
      cell.age = 0;
      cell.isDying = false;
    } else if (wasAlive && cell.alive) {
      cell.age++;
      cell.isDying = false;
    } else if (!cell.alive) {
      // It's dead now. If it was just killed, we keep the isDying flag for visual transition?
      // Actually, let's just clear it immediately for clean logic, but we pass the 'dying' color 
      // via the update logic in draw if we wanted a fade out. 
      // For this implementation, we will color based on current state.
      cell.isDying = false;
    }
  });

  state.generation++;
  updateStats();
}

// --- Rendering ---

function project(p, rotX, rotY, zoom) {
  // Rotate Y
  let x = p.x * Math.cos(rotY) - p.z * Math.sin(rotY);
  let z = p.z * Math.cos(rotY) + p.x * Math.sin(rotY);
  
  // Rotate X
  let y = p.y * Math.cos(rotX) - z * Math.sin(rotX);
  let zFinal = z * Math.cos(rotX) + p.y * Math.sin(rotX);
  
  // Perspective
  const fov = 900 * zoom;
  const scale = fov / (fov + zFinal);
  
  return {
    x: x * scale + width / 2,
    y: y * scale + height / 2,
    scale: scale,
    z: zFinal 
  };
}

function draw() {
  // Clear with trail effect
  // Using a very low opacity fill to create trails
  ctx.fillStyle = 'rgba(5, 5, 8, 0.25)'; 
  ctx.fillRect(0, 0, width, height);

  // Prepare projected cells
  const projectedCells = state.cells.map(cell => {
    const proj = project(cell.pos, state.rotation.x, state.rotation.y, state.zoom);
    return { cell, proj };
  });
  
  // Sort by Z (Painter's Algorithm)
  // Draw far cells first (higher Z)
  projectedCells.sort((a, b) => b.proj.z - a.proj.z);

  projectedCells.forEach(({ cell, proj }) => {
    // Cull backface slightly to give solid volume feel, but keep some transparency
    // This creates the "soft rear hemisphere" effect
    if (proj.z > CONFIG.sphereRadius * 0.9) return; 

    const size = CONFIG.cellSize * proj.scale;
    
    // Calculate Depth Lighting (Z-Fog)
    // Normalize Z from -R (front) to +R (back) -> 0 to 1
    // proj.z is camera space Z.
    // We want cells in front (negative Z) to be bright, cells at back (positive Z) to be dim.
    // But our projection math: zFinal > 0 is back.
    const depthRatio = (proj.z + CONFIG.sphereRadius) / (2 * CONFIG.sphereRadius); // 0 (front) to 1 (back)
    const alpha = 1 - (depthRatio * 0.85); // Fade out back cells more aggressively
    const brightness = 1 - (depthRatio * 0.65); // Dim back cells
    
    let colorStr;
    let shadowColor = "transparent";
    let shadowBlur = 0;

    if (cell.alive) {
      // Determine State Color
      let cBase;
      
      // Logic to ensure all 4 colors are visible
      // New (Cyan), Stable (Violet), Aging (Magenta)
      if (cell.age < 3) {
        cBase = CONFIG.colors.cyan; // New
      } else if (cell.age < 12) {
        cBase = CONFIG.colors.violet; // Stable
      } else {
        cBase = CONFIG.colors.magenta; // Aging
      }
      
      // Apply lighting adjustments
      const h = cBase.h;
      const s = cBase.s;
      const l = Math.min(90, cBase.l * brightness); // Lightness drops in back
      
      colorStr = `hsla(${h}, ${s}%, ${l}%, ${alpha})`;
      
      // Glow for front cells
      if (depthRatio < 0.7 && size > 1.5) {
        shadowColor = `hsla(${h}, ${s}%, 80%, ${alpha * 0.6})`;
        shadowBlur = 10 * proj.scale;
      }
    } else {
      // Dead cells - subtle grid
      colorStr = `rgba(40, 50, 70, ${alpha * 0.15})`;
    }
    
    // Handle Dying State (Amber)
    // We check the 'isDying' flag which was set in the logic step, 
    // but we need to persist it visually for a moment or handle it in the draw loop directly
    // The logic sets isDying=true on the cell that is dying.
    // We need to access the cell's state *before* it fully dies in the next frame.
    // Since stepGame updates state immediately, we rely on the fact that 
    // if a cell is dead, it might have just died. 
    // However, to make Amber visible, we need to color cells that are DEAD but were ALIVE in the previous frame.
    // Let's check a custom property or just handle the transition here.
    // Actually, the logic sets `cell.isDying = true` on the cell that is about to die.
    // But in `stepGame`, we apply updates immediately.
    // Let's modify the logic slightly: if a cell is dead, check if it has a 'dyingTimer' or similar.
    // For simplicity in this render loop, we will color based on the `isDying` flag 
    // which we will ensure is set for dead cells that just died.
    
    // Correction: The current logic sets `isDying` on the cell *before* it dies.
    // But then sets `alive=false`.
    // So here, if `!cell.alive` is true, we can check if it was recently dying?
    // Let's just use the flag we set: `cell.isDying`.
    // Wait, the logic sets `isDying` on the cell that is alive but will die?
    // No, the logic: `if (neighborCount < 2 || neighborCount > 3) { nextAlive = false; cell.isDying = true; }`
    // Then `updates.push({ isDying: !nextAlive && cell.alive })`
    // Then `cell.isDying = false` at the end.
    // So the flag is lost immediately.
    // Let's add a `deathTimer` to the cell to show Amber for a few frames.
    
    if (!cell.alive && cell.deathTimer > 0) {
      cell.deathTimer--;
      const cBase = CONFIG.colors.amber;
      const h = cBase.h;
      const s = cBase.s;
      const l = Math.min(90, cBase.l * brightness);
      colorStr = `hsla(${h}, ${s}%, ${l}%, ${alpha})`;
      shadowColor = `hsla(${h}, ${s}%, 80%, ${alpha * 0.5})`;
      shadowBlur = 8 * proj.scale;
    }
    
    ctx.fillStyle = colorStr;
    ctx.shadowColor = shadowColor;
    ctx.shadowBlur = shadowBlur;
    
    ctx.beginPath();
    ctx.arc(proj.x, proj.y, size, 0, Math.PI * 2);
    ctx.fill();
  });
  
  // Reset shadow
  ctx.shadowBlur = 0;
  
  // Draw Atmosphere/Rim Light
  // This is a static gradient overlay to give the sphere a "3D object" feel
  const cx = width / 2;
  const cy = height / 2;
  const r = CONFIG.sphereRadius * state.zoom;
  
  // Rim light gradient
  const rimGrad = ctx.createRadialGradient(cx, cy, r * 0.85, cx, cy, r * 1.05);
  rimGrad.addColorStop(0, 'rgba(0,0,0,0)');
  rimGrad.addColorStop(0.7, 'rgba(0,0,0,0)');
  // Cyan tint on the edge
  rimGrad.addColorStop(1, 'rgba(0, 240, 255, 0.12)');
  
  ctx.fillStyle = rimGrad;
  ctx.beginPath();
  ctx.arc(cx, cy, r * 1.1, 0, Math.PI * 2);
  ctx.fill();
  
  // Inner shadow for depth
  const innerGrad = ctx.createRadialGradient(cx, cy, r * 0.1, cx, cy, r);
  innerGrad.addColorStop(0, 'rgba(0,0,0,0)');
  innerGrad.addColorStop(1, 'rgba(0,0,0,0.6)');
  
  ctx.fillStyle = innerGrad;
  ctx.beginPath();
  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  ctx.fill();
}

// --- Main Loop ---

function loop(timestamp) {
  if (state.isRunning) {
    // Auto rotation
    state.rotation.y += CONFIG.rotationSpeed;
    
    // Update Logic based on speed
    if (timestamp - state.lastFrameTime > (1000 / state.speed)) {
      stepGame();
      state.lastFrameTime = timestamp;
    }
  }
  
  draw();
  requestAnimationFrame(loop);
}

// --- Interaction ---

function updateStats() {
  document.getElementById('gen-count').innerText = state.generation;
  const pop = state.cells.filter(c => c.alive).length;
  document.getElementById('pop-count').innerText = pop;
}

    e.target.classList.add('primary');
    e.target.classList.remove('secondary');
  } else {
    e.target.innerText = "Resume";
    e.target.classList.remove('primary');
    e.target.classList.add('secondary');
  }

  initGame();
});

  state.isRunning = false;
  btnPause.innerText = "Resume";
  btnPause.classList.remove('primary');
  btnPause.classList.add('secondary');
  stepGame();
  draw();
});

document.getElementById('speed-slider').addEventListener('input', (e) => {
  state.speed = parseInt(e.target.value);
});

// Mouse Drag Rotation
const container = document.getElementById('canvas-container');

container.addEventListener('mousedown', (e) => {
  state.isDragging = true;
  state.lastMouse = { x: e.clientX, y: e.clientY };
});

window.addEventListener('mousemove', (e) => {
  if (!state.isDragging) return;
  const deltaX = e.clientX - state.lastMouse.x;
  const deltaY = e.clientY - state.lastMouse.y;
  
  state.rotation.y += deltaX * 0.005;
  state.rotation.x += deltaY * 0.005;
  
  state.lastMouse = { x: e.clientX, y: e.clientY };
});

window.addEventListener('mouseup', () => {
  state.isDragging = false;
});

// Zoom
container.addEventListener('wheel', (e) => {
  e.preventDefault();
  const zoomSensitivity = 0.001;
  state.zoom -= e.deltaY * zoomSensitivity;
  state.zoom = Math.max(0.5, Math.min(3, state.zoom));
});

// --- Init ---
initGame();
requestAnimationFrame(loop);
