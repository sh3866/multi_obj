<script>
    /**
     * VISUAL ENGINE
     * Draws the retro pixel art scene procedurally.
     */
    const canvas = document.getElementById('world');
    const ctx = canvas.getContext('2d');

    // Pixel Scale
    const PIXEL_SIZE = 4;
    
    // State
    let width, height;
    let frame = 0;
    let trees = [];
    let snowParticles = [];
    let skierOffset = 0;
    
    // Colors
    const C_SKY_TOP = '#87CEEB';
    const C_SKY_BOT = '#E0F7FA';
    const C_SNOW = '#FFFFFF';
    const C_SNOW_DARK = '#D4E6F1';
    const C_TREE_DARK = '#1B5E20';
    const C_TREE_LIGHT = '#2E7D32';

    // Assets (Drawn via functions)
    
    function drawPixelSkier(x, y, scale) {
        ctx.save();
        ctx.translate(x, y);
        
        // Skier Palette
        const cSkin = '#FFCC80';
        const cHat = '#D32F2F';
        const cJacket = '#D32F2F';
        const cPants = '#1A237E';
        const cSkis = '#FBC02D';
        const cPoles = '#555555';

        // Bobbing animation
        const bob = Math.sin(frame * 0.2) * 2;
        const lean = Math.sin(frame * 0.1) * 0.05; // Subtle lean

        ctx.translate(0, bob);
        ctx.rotate(lean);

        const p = scale; // pixel unit for this sprite

        // Skis (drawn first to be behind)
        ctx.fillStyle = cSkis;
        ctx.fillRect(-4*p, 10*p, p, 14*p); // Left ski
        ctx.fillRect(3*p, 10*p, p, 14*p);  // Right ski

        // Poles
        ctx.fillStyle = cPoles;
        ctx.fillRect(-5*p, 4*p, p, 16*p);
        ctx.fillRect(4*p, 4*p, p, 16*p);

        // Legs
        ctx.fillStyle = cPants;
        ctx.fillRect(-3*p, 8*p, 2*p, 4*p);
        ctx.fillRect(2*p, 8*p, 2*p, 4*p);

        // Body
        ctx.fillStyle = cJacket;
        ctx.fillRect(-3*p, 2*p, 6*p, 7*p);

        // Head
        ctx.fillStyle = cSkin;
        ctx.fillRect(-2*p, -2*p, 4*p, 4*p);
        
        // Hat
        ctx.fillStyle = cHat;
        ctx.fillRect(-2*p, -3*p, 4*p, 2*p);

        ctx.restore();
    }

    function drawTree(x, y, scale) {
        ctx.save();
        ctx.translate(x, y);
        
        // Shadow
        ctx.fillStyle = 'rgba(0,0,0,0.1)';
        ctx.fillRect(-4*scale, 20*scale, 8*scale, 2*scale);

        const p = scale;
        
        // Trunk
        ctx.fillStyle = '#5D4037';
        ctx.fillRect(-p, 14*p, 2*p, 6*p);

        // Leaves (Simple Triangle-ish shape)
        ctx.fillStyle = C_TREE_DARK;
        
        // Layer 1 (Bottom)
        ctx.fillRect(-6*p, 6*p, 12*p, 4*p);
        ctx.fillRect(-5*p, 4*p, 10*p, 2*p);
        
        // Layer 2 (Mid)
        ctx.fillRect(-5*p, 2*p, 10*p, 4*p);
        
        // Layer 3 (Top)
        ctx.fillRect(-3*p, -2*p, 6*p, 4*p);

        // Highlights (Lighter green pixels)
        ctx.fillStyle = C_TREE_LIGHT;
        ctx.fillRect(-4*p, 6*p, 2*p, 2*p);
        ctx.fillRect(-3*p, 2*p, 2*p, 2*p);

        ctx.restore();
    }

    function drawSnowflake(x, y, size) {
        ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
        ctx.fillRect(x, y, size, size);
    }

    // Initialization
    function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        
        // Set internal resolution lower for pixel look, then scale up via CSS
        // But for crisp rendering on modern screens, we draw 1:1 and use CSS image-rendering
        canvas.width = width;
        canvas.height = height;
        
        initWorld();
    }

    function initWorld() {
        trees = [];
        // Create a field of trees
        // We want them to look like they are on a slope.
        // Perspective: Trees at bottom are bigger, trees at top are smaller.
        
        const rows = 20;
        const spacingX = 100;
        
        for(let i=0; i<rows; i++) {
            // Z-depth simulation (0 to 1, 1 is far)
            let depth = i / rows;
            // Invert for drawing order (back to front)
            let rowIdx = rows - i; 
            
            // Perspective math
            let perspective = 1 - (rowIdx * 0.04); // 1.0 at bottom, 0.2 at top
            
            let y = height - (rowIdx * (height/rows)) - 50; // Distribute vertically
            // Curve the slope slightly
            y += Math.sin(rowIdx * 0.5) * 20;
            
            let scale = 1 + (rowIdx * 0.15); // Scale increases as we get closer
            
            // Add random trees in this row
            let numTrees = Math.floor(Math.random() * 3) + 1;
            for(let j=0; j<numTrees; j++) {
                let x = Math.random() * width;
                trees.push({x, y, scale, z: rowIdx});
            }
        }
        
        // Sort trees by Z (distance) so far trees draw first
        trees.sort((a, b) => a.z - b.z);

        // Snow particles
        snowParticles = [];
        for(let i=0; i<100; i++) {
            snowParticles.push({
                x: Math.random() * width,
                y: Math.random() * height,
                speed: Math.random() * 2 + 1,
                size: Math.random() * 3 + 1
            });
        }
    }

    function drawBackground() {
        // Sky Gradient
        let grad = ctx.createLinearGradient(0, 0, 0, height);
        grad.addColorStop(0, C_SKY_TOP);
        grad.addColorStop(1, C_SKY_BOT);
        ctx.fillStyle = grad;
        ctx.fillRect(0, 0, width, height);

        // Snowy Slope (Bottom half)
        // Simple geometric slope
        ctx.beginPath();
        ctx.moveTo(0, height);
        ctx.lineTo(0, height * 0.4);
        ctx.bezierCurveTo(width * 0.3, height * 0.3, width * 0.7, height * 0.5, width, height * 0.4);
        ctx.lineTo(width, height);
        ctx.fillStyle = C_SNOW;
        ctx.fill();

        // Slope shading (simple shadow on the right side)
        ctx.beginPath();
        ctx.moveTo(width * 0.7, height * 0.5);
        ctx.lineTo(width, height * 0.4);
        ctx.lineTo(width, height);
        ctx.lineTo(width * 0.7, height);
        ctx.fillStyle = C_SNOW_DARK;
        ctx.globalAlpha = 0.5;
        ctx.fill();
        ctx.globalAlpha = 1.0;
    }

    function drawMountains() {
        ctx.fillStyle = '#B0BEC5'; // Distant mountains
        ctx.beginPath();
        ctx.moveTo(0, height * 0.5);
        ctx.lineTo(100, height * 0.35);
        ctx.lineTo(200, height * 0.45);
        ctx.lineTo(350, height * 0.25);
        ctx.lineTo(500, height * 0.4);
        ctx.lineTo(width, height * 0.3);
        ctx.lineTo(width, height * 0.5);
        ctx.lineTo(0, height * 0.5);
        ctx.fill();
    }

    function animate() {
        ctx.clearRect(0, 0, width, height);

        drawBackground();
        drawMountains();

        // Draw Trees
        trees.forEach(tree => {
            drawTree(tree.x, tree.y, tree.scale);
        });

        // Draw Skier
        // Skier stays roughly at bottom center, maybe moves slightly
        const skierX = width / 2;
        const skierY = height - 180; // Just above the UI
        drawPixelSkier(skierX, skierY, PIXEL_SIZE);

        // Draw Snow Particles
        snowParticles.forEach(p => {
            p.y += p.speed;
            p.x -= 1; // Wind blowing left
            if (p.y > height) p.y = 0;
            if (p.x < 0) p.x = width;
            drawSnowflake(p.x, p.y, p.size);
        });

        frame++;
        requestAnimationFrame(animate);
    }

    // Event Listeners
    window.addEventListener('resize', resize);
    
    // Button Interaction
    document.getElementById('startBtn').addEventListener('click', function() {
        this.innerText = "LOADING...";
        this.style.background = "#fff";
        setTimeout(() => {
            alert("Game Start! (This is a visual demo)");
            this.innerText = "START GAME";
            this.style.background = "var(--ui-accent)";
        }, 500);
    });

    // Init
    resize();
    animate();

</script>
