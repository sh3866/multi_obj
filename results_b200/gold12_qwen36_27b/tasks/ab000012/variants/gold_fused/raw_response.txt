<script>
    // --- Background Particle System ---
    const canvas = document.getElementById('canvas-bg');
    const ctx = canvas.getContext('2d');

    let width, height;
    let particles = [];
    
    // Altar position for particles to orbit
    let centerX, centerY;

    function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;
        
        // Update center coordinates
        centerX = width / 2;
        centerY = (height / 2) - 50; // Approximate center of the altar
    }

    class Particle {
        constructor() {
            this.reset(true);
        }

        reset(initial = false) {
            // Spawn around the center altar area
            this.angle = Math.random() * Math.PI * 2;
            this.radius = Math.random() * 200 + 100; // Distance from center
            
            this.x = centerX + Math.cos(this.angle) * this.radius;
            this.y = centerY + Math.sin(this.angle) * this.radius;
            
            this.vx = (Math.random() - 0.5) * 0.2;
            this.vy = -(Math.random() * 0.5 + 0.1); // Drift up
            
            this.size = Math.random() * 1.5;
            this.alpha = Math.random() * 0.5;
            this.life = 1;
            this.decay = Math.random() * 0.005 + 0.002;
            
            // Color: mix of Gold and Teal
            this.isGold = Math.random() > 0.6;
            this.color = this.isGold ? `212, 175, 55` : `79, 209, 197`;
        }

        update() {
            this.x += this.vx;
            this.y += this.vy;
            this.life -= this.decay;

            // Spiral effect
            this.angle += 0.002;
            
            if (this.life <= 0) {
                this.reset();
            }
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(${this.color}, ${this.life * this.alpha})`;
            ctx.fill();
        }
    }

    function initParticles() {
        particles = [];
        const count = Math.floor((width * height) / 12000); 
        for (let i = 0; i < count; i++) {
            particles.push(new Particle());
        }
    }

    function animate() {
        ctx.clearRect(0, 0, width, height);
        
        // Subtle connecting lines near center
        ctx.lineWidth = 0.5;
        for (let i = 0; i < particles.length; i++) {
            particles[i].update();
            particles[i].draw();
            
            // Connect close particles (only if close to center to save perf)
            if (i % 5 === 0) { // Optimization: check fewer particles
                const dx = particles[i].x - centerX;
                const dy = particles[i].y - centerY;
                const dist = Math.sqrt(dx * dx + dy * dy);
                
                if (dist < 300) {
                    for (let j = i + 1; j < particles.length; j++) {
                        if (j % 5 !== 0) continue;
                        const dx2 = particles[i].x - particles[j].x;
                        const dy2 = particles[i].y - particles[j].y;
                        const dist2 = Math.sqrt(dx2 * dx2 + dy2 * dy2);
                        
                        if (dist2 < 80) {
                            ctx.beginPath();
                            ctx.strokeStyle = `rgba(79, 209, 197, ${0.05 * (1 - dist2/80)})`;
                            ctx.moveTo(particles[i].x, particles[i].y);
                            ctx.lineTo(particles[j].x, particles[j].y);
                            ctx.stroke();
                        }
                    }
                }
            }
        }
        requestAnimationFrame(animate);
    }

    // --- Interactions ---
    window.addEventListener('resize', () => {
        resize();
        initParticles();
    });

    function interact(action) {
        const btn = event.currentTarget;
        
        // Button press effect
        btn.style.transform = 'scale(0.96)';
        btn.style.filter = 'brightness(1.2)';
        
        setTimeout(() => {
            btn.style.transform = '';
            btn.style.filter = '';
        }, 150);
        
        // Flash the altar
        const orb = document.querySelector('.orb-core');
        orb.style.boxShadow = '0 0 60px #fff';
        setTimeout(() => {
            orb.style.boxShadow = '';
        }, 200);
        
        console.log(`Action triggered: ${action}`);
    }

    // Initialize
    resize();
    initParticles();
    animate();

</script>
