<script>
    // --- Background Particle System (Dust & Embers) ---
    const canvas = document.getElementById('canvas-bg');
    const ctx = canvas.getContext('2d');

    let width, height;
    let particles = [];

    function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;
    }

    class Particle {
        constructor() {
            this.reset();
        }

        reset() {
            this.x = Math.random() * width;
            this.y = height + Math.random() * 100;
            this.vx = (Math.random() - 0.5) * 0.3; // Slower drift
            this.vy = -(Math.random() * 0.8 + 0.2); // Slow rise
            this.size = Math.random() * 1.5 + 0.5;
            this.alpha = Math.random() * 0.4;
            this.life = Math.random() * 0.5 + 0.5;
            
            // 10% chance to be gold, 90% white/teal dust
            const isGold = Math.random() > 0.9;
            if (isGold) {
                this.color = '240, 192, 90'; // Gold
            } else {
                this.color = '200, 230, 255'; // Cool Dust
            }
        }

        update() {
            this.x += this.vx;
            this.y += this.vy;
            
            // Fade in/out
            if (this.y < height * 0.2) {
                this.alpha -= 0.002;
            } else if (this.y > height * 0.8) {
                this.alpha += 0.002;
            }

            if (this.alpha <= 0 || this.y < -10) {
                this.reset();
            }
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(${this.color}, ${this.alpha})`;
            ctx.fill();
        }
    }

    function initParticles() {
        particles = [];
        const count = Math.floor((width * height) / 20000); // Slightly sparser for elegance
        for (let i = 0; i < count; i++) {
            particles.push(new Particle());
        }
    }

    function animate() {
        ctx.clearRect(0, 0, width, height);
        
        for (let i = 0; i < particles.length; i++) {
            particles[i].update();
            particles[i].draw();
        }
        requestAnimationFrame(animate);
    }

    // --- Interactions ---
    window.addEventListener('resize', () => {
        resize();
        initParticles();
    });

    function interact(action) {
        const btn = event.currentTarget;
        // Simple scale animation
        btn.style.transform = 'scale(0.96)';
        setTimeout(() => {
            btn.style.transform = '';
        }, 150);
        
        // Logic placeholder
        console.log(`Navigating to: ${action}`);
    }

    // Initialize
    resize();
    initParticles();
    animate();

</script>
