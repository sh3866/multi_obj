<script>
    // Simple interaction logic
    function switchTab(tabName) {
        // Update Tabs
        const tabs = document.querySelectorAll('.log-tab');
        tabs.forEach(t => t.classList.remove('active'));
        event.target.classList.add('active');

        // Update Content
        document.getElementById('quest-content').style.display = tabName === 'quests' ? 'block' : 'none';
        document.getElementById('clue-content').style.display = tabName === 'clues' ? 'block' : 'none';
    }

    // Roster Selection Logic
    const npcCards = document.querySelectorAll('.npc-card');
    npcCards.forEach(card => {
        card.addEventListener('click', () => {
            npcCards.forEach(c => c.classList.remove('active'));
            card.classList.add('active');
            
            // In a real app, this would update the main stage content
            // For visual demo, we just add a subtle animation to the stage
            const sprite = document.querySelector('.npc-sprite');
            sprite.style.opacity = '0';
            sprite.style.transform = 'scale(0.95) translateY(20px)';
            setTimeout(() => {
                sprite.style.opacity = '1';
                sprite.style.transform = 'scale(1) translateY(0)';
            }, 150);
        });
    });

    // Choice Button Hover Effect Logic
    const choices = document.querySelectorAll('.choice-btn');
    choices.forEach(btn => {
        btn.addEventListener('mouseenter', () => {
            // Visual feedback is handled by CSS
        });
    });

    // Add a gentle breathing animation to the active NPC
    const sprite = document.querySelector('.npc-sprite');
    let isHovering = false;
    
    sprite.addEventListener('mouseenter', () => isHovering = true);
    sprite.addEventListener('mouseleave', () => isHovering = false);

    function animateSprite() {
        if (!isHovering) {
            const time = Date.now() * 0.0015;
            const y = Math.sin(time) * 4;
            sprite.style.transform = `translateY(${y}px)`;
        }
        requestAnimationFrame(animateSprite);
    }
    animateSprite();

</script>
