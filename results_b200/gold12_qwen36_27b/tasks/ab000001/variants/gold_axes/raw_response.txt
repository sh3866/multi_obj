    <script>
        // --- Game Logic & Drawing ---

        const GRID_SIZE = 6;
        
        // Tile Types (SVG Paths)
        // Coordinates are 0-100, center is 50,50
        const TILE_TYPES = {
            EMPTY: { paths: [] },
            LINE: { paths: ['M50 0 V100'], con: [1,0,1,0] }, // Vertical
            LINE_H: { paths: ['M0 50 H100'], con: [0,1,0,1] }, // Horizontal
            CORNER_TR: { paths: ['M50 0 V50 H100'], con: [1,1,0,0] },
            CORNER_BR: { paths: ['M100 50 V100 H50'], con: [0,1,1,0] },
            CORNER_BL: { paths: ['M0 50 V100 H50'], con: [0,0,1,1] },
            CORNER_TL: { paths: ['M50 0 V50 H0'], con: [1,0,0,1] },
            T_UP: { paths: ['M0 50 H100 M50 50 V100'], con: [0,1,1,1] },
            T_RIGHT: { paths: ['M0 50 H100 M50 50 V0'], con: [1,0,1,1] },
            T_LEFT: { paths: ['M0 50 H100 M50 50 V0'], con: [1,1,1,0] }
        };

        class Game {
            constructor() {
                this.gridEl = document.getElementById('grid');
                this.cubEl = document.getElementById('cub');
                this.starEl = document.getElementById('star');
                this.overlay = document.getElementById('winOverlay');
                this.cells = []; 
                this.startPos = {x: 0, y: 0};
                this.endPos = {x: 5, y: 4}; 
                this.cubPos = {x: 0, y: 0};
                
                // Hardcoded layout for visual consistency
                this.layout = [
                    ['C_TL', 'L', 'C_TR', 'L', 'C_TL', 'L'],
                    ['L', 'C_BR', 'L', 'C_TL', 'L', 'L'],
                    ['L', 'L', 'C_BL', 'C_BR', 'C_BL', 'C_BR'],
                    ['L', 'C_TL', 'L', 'L', 'C_BL', 'L'],
                    ['C_TR', 'L', 'C_BL', 'C_BR', 'L', 'L'],
                    ['L', 'L', 'L', 'L', 'L', 'L']
                ];

                this.init();
            }

            init() {
                this.renderBoard();
                // Initial positions
                this.updateTokenPosition(this.startPos.x, this.startPos.y);
                this.updateStarPosition(this.endPos.x, this.endPos.y);
            }

            renderBoard() {
                this.gridEl.innerHTML = '';
                this.cells = [];

                for(let y=0; y<GRID_SIZE; y++) {
                    let row = [];
                    for(let x=0; x<GRID_SIZE; x++) {
                        const key = this.layout[y][x];
                        const tileType = this.getTileType(key);
                        
                        const div = document.createElement('div');
                        div.className = 'tile';
                        div.innerHTML = this.createSVG(tileType.paths);
                        
                        div.dataset.x = x;
                        div.dataset.y = y;
                        
                        this.gridEl.appendChild(div);
                        row.push(div);
                    }
                    this.cells.push(row);
                }
            }

            getTileType(key) {
                const map = {
                    'L': TILE_TYPES.LINE,
                    'C_TL': TILE_TYPES.CORNER_TL,
                    'C_TR': TILE_TYPES.CORNER_TR,
                    'C_BL': TILE_TYPES.CORNER_BL,
                    'C_BR': TILE_TYPES.CORNER_BR,
                    'C': TILE_TYPES.CORNER_TL 
                };
                return map[key] || TILE_TYPES.EMPTY;
            }

            createSVG(paths) {
                let svgContent = paths.map(p => `<path d="${p}" />`).join('');
                return `<svg viewBox="0 0 100 100" preserveAspectRatio="none">${svgContent}</svg>`;
            }

            updateTokenPosition(x, y) {
                const p = 100 / GRID_SIZE;
                const left = (x * p) + (p / 2);
                const top = (y * p) + (p / 2);
                this.cubEl.style.left = `${left}%`;
                this.cubEl.style.top = `${top}%`;
            }

            updateStarPosition(x, y) {
                const p = 100 / GRID_SIZE;
                const left = (x * p) + (p / 2);
                const top = (y * p) + (p / 2);
                this.starEl.style.left = `${left}%`;
                this.starEl.style.top = `${top}%`;
            }

            reset() {
                this.overlay.classList.remove('active');
                document.querySelectorAll('.tile').forEach(t => t.classList.remove('active'));
                this.cubPos = {x: 0, y: 0};
                this.updateTokenPosition(0, 0);
            }

            async run() {
                // Path to Star at 5,4
                const pathToStar = [
                    {x:0, y:0}, {x:0, y:1}, {x:0, y:2}, {x:0, y:3}, {x:0, y:4}, {x:0, y:5},
                    {x:1, y:5}, {x:2, y:5}, {x:3, y:5}, {x:4, y:5}, {x:4, y:4}, {x:4, y:3}, 
                    {x:5, y:3}, {x:5, y:4} 
                ];

                for (let i = 0; i < pathToStar.length; i++) {
                    const step = pathToStar[i];
                    const tile = this.cells[step.y][step.x];
                    
                    // Highlight
                    tile.classList.add('active');
                    
                    // Move
                    this.updateTokenPosition(step.x, step.y);
                    
                    // Wait
                    await new Promise(r => setTimeout(r, 350));
                }

                setTimeout(() => {
                    this.overlay.classList.add('active');
                }, 300);
            }
        }

        const game = new Game();

    </script>
