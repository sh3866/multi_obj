    <script>
        // --- Game Logic & Drawing ---

        const GRID_SIZE = 6;
        
        // Tile Types (SVG Paths)
        // Coordinates 0-100. Center is 50,50.
        const TILE_TYPES = {
            LINE: { paths: ['M50 0 V100'] }, // Vertical
            LINE_H: { paths: ['M0 50 H100'] }, // Horizontal
            CORNER_TR: { paths: ['M50 0 V50 H100'] },
            CORNER_BR: { paths: ['M100 50 V100 H50'] },
            CORNER_BL: { paths: ['M0 50 V100 H50'] },
            CORNER_TL: { paths: ['M50 0 V50 H0'] },
            EMPTY: { paths: [] }
        };

        // Layout map
        // C: Corner, L: Line
        const PUZZLE_MAP = [
            ['C_TL', 'L', 'C_TR', 'L', 'C_TL', 'L'],
            ['L', 'C_BR', 'L', 'C_TL', 'L', 'L'],
            ['L', 'L', 'C_BL', 'C_BR', 'C_BL', 'C_BR'],
            ['L', 'C_TL', 'L', 'L', 'C_BL', 'L'],
            ['C_TR', 'L', 'C_BL', 'C_BR', 'L', 'L'],
            ['L', 'L', 'L', 'L', 'L', 'L']
        ];

        function createSVG(paths) {
            let svgContent = paths.map(p => `<path d="${p}" />`).join('');
            return `<svg viewBox="0 0 100 100" preserveAspectRatio="none">${svgContent}</svg>`;
        }

        class Game {
            constructor() {
                this.gridEl = document.getElementById('grid');
                this.cubEl = document.getElementById('cub');
                this.starEl = document.getElementById('star');
                this.overlay = document.getElementById('winOverlay');
                this.cells = []; 
                this.startPos = {x: 0, y: 0};
                this.endPos = {x: 5, y: 4}; 
                this.cubPos = {x: 0, y: 0};
                
                this.init();
            }

            init() {
                this.renderBoard();
                this.updateTokenPosition(0, 0);
                this.updateStarPosition(this.endPos.x, this.endPos.y);
            }

            renderBoard() {
                this.gridEl.innerHTML = '';
                this.cells = [];

                for(let y=0; y<GRID_SIZE; y++) {
                    let row = [];
                    for(let x=0; x<GRID_SIZE; x++) {
                        const key = PUZZLE_MAP[y][x];
                        const tileType = TILE_TYPES[key] || TILE_TYPES.EMPTY;
                        
                        const div = document.createElement('div');
                        div.className = 'tile';
                        div.innerHTML = createSVG(tileType.paths);
                        
                        div.dataset.x = x;
                        div.dataset.y = y;
                        
                        this.gridEl.appendChild(div);
                        row.push(div);
                    }
                    this.cells.push(row);
                }
            }

            updateTokenPosition(x, y) {
                // Grid is 6x6. 
                // We need to position relative to the .token-layer which matches grid size.
                // Each cell is 100/6 %. Center is (index * 100/6) + (50/6)%
                const cellPerc = 100 / GRID_SIZE;
                const centerOffset = cellPerc / 2;
                
                const left = (x * cellPerc) + centerOffset;
                const top = (y * cellPerc) + centerOffset;
                
                this.cubEl.style.left = `${left}%`;
                this.cubEl.style.top = `${top}%`;
            }

            updateStarPosition(x, y) {
                const cellPerc = 100 / GRID_SIZE;
                const centerOffset = cellPerc / 2;
                const left = (x * cellPerc) + centerOffset;
                const top = (y * cellPerc) + centerOffset;
                this.starEl.style.left = `${left}%`;
                this.starEl.style.top = `${top}%`;
            }

            reset() {
                this.overlay.classList.remove('active');
                document.querySelectorAll('.tile svg').forEach(s => s.style.stroke = 'var(--path-ink)');
                this.cubPos = {x: 0, y: 0};
                this.updateTokenPosition(0, 0);
            }

            async run() {
                // Hardcoded path to Star at 5,4
                const pathToStar = [
                    {x:0, y:0}, {x:0, y:1}, {x:0, y:2}, {x:0, y:3}, {x:0, y:4}, {x:0, y:5},
                    {x:1, y:5}, {x:2, y:5}, {x:3, y:5}, {x:4, y:5}, {x:4, y:4}, {x:4, y:3}, 
                    {x:5, y:3}, {x:5, y:4} 
                ];

                for (let i = 0; i < pathToStar.length; i++) {
                    const step = pathToStar[i];
                    
                    const tile = this.cells[step.y][step.x];
                    tile.querySelector('svg').style.stroke = 'var(--path-active)';
                    
                    this.updateTokenPosition(step.x, step.y);
                    
                    await new Promise(r => setTimeout(r, 400));
                }

                setTimeout(() => {
                    this.overlay.classList.add('active');
                }, 200);
            }
        }

        const game = new Game();

    </script>
