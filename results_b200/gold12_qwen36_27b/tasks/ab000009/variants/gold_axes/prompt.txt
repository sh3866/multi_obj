        boardEl.innerHTML = '';
        
        for (let r = 0; r < 8; r++) {
            for (let c = 0; c < 8; c++) {
                const square = document.createElement('div');
                const isLight = (r + c) % 2 === 0;
                square.className = `square ${isLight ? 'light' : 'dark'}`;
                square.dataset.row = r;
                square.dataset.col = c;
                square.onclick = () => handleSquareClick(r, c);

                // Coordinates
                if (c === 0) {
                    const rank = document.createElement('span');
                    rank.className = 'coord rank';
                    rank.innerText = 8 - r;
                    square.appendChild(rank);
                }
                if (r === 7) {
                    const file = document.createElement('span');
                    file.className = 'coord file';
                    file.innerText = String.fromCharCode(97 + c);
                    square.appendChild(file);
                }

                // Piece
                const pieceCode = board[r][c];
                if (pieceCode) {
                    const piece = document.createElement('div');
                    piece.className = 'piece';
                    piece.innerHTML = SVG_PIECES[pieceCode];
                    square.appendChild(piece);
                }

                // Selection highlight
                if (selectedSquare && selectedSquare.r === r && selectedSquare.c === c) {
                    square.classList.add('selected');
                }

                boardEl.appendChild(square);
            }
        }
    }

    function updateStatus() {
        const player = turn === 'w' ? 'White' : 'Black';
        statusText.innerText = `${player} to Move`;
        turnDot.className = `turn-dot ${turn === 'w' ? 'white' : 'black'}`;
    }

    // --- Interaction ---
    function handleSquareClick(r, c) {
        const piece = board[r][c];
        const isOwnPiece = piece && piece.startsWith(turn);

        if (isOwnPiece) {
            // Select piece
            selectedSquare = { r, c };
            renderBoard();
        } else if (selectedSquare) {
            // Move piece (Visual only for this demo)
            movePiece(selectedSquare.r, selectedSquare.c, r, c);
        }
    }

    function movePiece(fromR, fromC, toR, toC) {
        const piece = board[fromR][fromC];
        
        // Update board state
        board[toR][toC] = piece;
        board[fromR][fromC] = '';
        
        // Add fake move to history
        addMoveToHistory(piece, fromR, fromC, toR, toC);

        // Switch turn
        turn = turn === 'w' ? 'b' : 'w';
        selectedSquare = null;
        
        renderBoard();
        updateStatus();
    }

    function addMoveToHistory(piece, fromR, fromC, toR, toC) {
        const files = 'abcdefgh';
        const ranks = '87654321';
        
        const notation = (piece[1] === 'P' ? '' : piece[1].toUpperCase()) + 
                         files[toC] + ranks[toR];

        const moveNum = Math.ceil((moveHistoryEl.children.length + 1) / 2);
        
        // If it's white's move, create new row
        if (turn === 'b') { // Turn is already switched here, so if it was 'b', now it's 'w' (Wait, logic check)
             // If current turn variable is 'b', that means White just moved.
             // Wait, I switched turn at end of function. 
             // Let's check logic: 
             // If turn was 'w', user moved white. Now turn is 'b'. This was a white move.
        }

        if (turn === 'b') { // White just moved
            const row = document.createElement('div');
            row.className = 'move-row';
            row.innerHTML = `
                <div class="move-num">${moveNum}.</div>
                <div class="move-white">${notation}</div>
                <div class="move-black"></div>
            `;
            moveHistoryEl.appendChild(row);
        } else { // Black just moved
            const lastRow = moveHistoryEl.lastElementChild;
            if (lastRow) {
                lastRow.querySelector('.move-black').innerText = notation;
            }
        }
        
        // Auto scroll
        moveHistoryEl.scrollTop = moveHistoryEl.scrollHeight;
    }

    function resetGame() {
        board = JSON.parse(JSON.stringify(initialBoard));
        turn = 'w';
        selectedSquare = null;
        moveHistoryEl.innerHTML = '';
        renderBoard();
        updateStatus();
    }

    // --- Init ---
    renderBoard();
    updateStatus();

</script>
