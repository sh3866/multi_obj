    /* --- Logic --- */
    function showTooltip(e, skill) {
        const ttTitle = document.getElementById('tt-title');
        const ttType = document.getElementById('tt-type');
        const ttDesc = document.getElementById('tt-desc');
        const ttReq = document.getElementById('tt-req');

        ttTitle.textContent = skill.name;
        ttType.textContent = skill.type;
        ttDesc.textContent = skill.desc;
        
        // Reset styles
        ttReq.classList.remove('locked');
        
        if (skill.unlocked) {
            ttReq.textContent = "UNLOCKED";
            ttReq.style.color = "var(--gold)";
        } else {
            const reqSkill = skills.find(s => s.id === skill.req);
            ttReq.textContent = `Requires: ${reqSkill ? reqSkill.name : 'None'}`;
            ttReq.classList.add('locked');
        }

        tooltip.classList.add('visible');
        moveTooltip(e);
    }

    function moveTooltip(e) {
        // Keep tooltip on screen
        let x = e.clientX + 20;
        let y = e.clientY + 20;
        
        if (x + 320 > window.innerWidth) x = e.clientX - 340;
        if (y + 200 > window.innerHeight) y = e.clientY - 200;

        tooltip.style.left = `${x}px`;
        tooltip.style.top = `${y}px`;
    }

    function hideTooltip() {
        tooltip.classList.remove('visible');
    }

    function toggleSkill(id) {
        const skill = skills.find(s => s.id === id);
        if (skill.unlocked) return; // Already unlocked

        // Check availability
        const parent = skills.find(s => s.id === skill.req);
        if (!parent || !parent.unlocked) return; // Cannot unlock yet

        // Unlock logic (Visual only)
        skill.unlocked = true;
        
        // Update Node
        const nodeEl = document.querySelector(`.node[data-id="${id}"]`);
        nodeEl.classList.remove('available', 'locked');
        nodeEl.classList.add('unlocked');

        // Update Line
        const lineEl = document.getElementById(`conn-${id}`);
        if (lineEl) {
            lineEl.classList.remove('available');
            lineEl.classList.add('active');
        }

        // Check children availability
        skills.forEach(child => {
            if (child.req === id && !child.unlocked) {
                const childNode = document.querySelector(`.node[data-id="${child.id}"]`);
                const childLine = document.getElementById(`conn-${child.id}`);
                
                childNode.classList.remove('locked');
                childNode.classList.add('available');
                
                if(childLine) {
                    childLine.classList.remove('locked');
                    childLine.classList.add('available');
                }
            }
        });
    }

    /* --- Pan & Zoom System --- */
    let isPanning = false;
    let startX, startY;
    let panX = -300; // Initial offset
    let panY = -100;
    let scale = 1;

    viewport.addEventListener('mousedown', (e) => {
        if (e.target.closest('.node')) return;
        isPanning = true;
        startX = e.clientX - panX;
        startY = e.clientY - panY;
        viewport.style.cursor = 'grabbing';
    });

    window.addEventListener('mouseup', () => {
        isPanning = false;
        viewport.style.cursor = 'grab';
    });

    window.addEventListener('mousemove', (e) => {
        if (!isPanning) return;
        e.preventDefault();
        panX = e.clientX - startX;
        panY = e.clientY - startY;
        updateTransform();
    });

    viewport.addEventListener('wheel', (e) => {
        e.preventDefault();
        const zoomSensitivity = 0.001;
        const delta = -e.deltaY * zoomSensitivity;
        const newScale = Math.min(Math.max(0.5, scale + delta), 2);
        
        // Zoom towards mouse pointer logic
        const rect = viewport.getBoundingClientRect();
        const mouseX = e.clientX - rect.left;
        const mouseY = e.clientY - rect.top;
        
        // Calculate offset adjustment
        const scaleChange = newScale - scale;
        panX -= mouseX * scaleChange;
        panY -= mouseY * scaleChange;
        
        scale = newScale;
        updateTransform();
    });

    function updateTransform() {
        world.style.transform = `translate(${panX}px, ${panY}px) scale(${scale})`;
    }

    // Start
    init();

</script>
