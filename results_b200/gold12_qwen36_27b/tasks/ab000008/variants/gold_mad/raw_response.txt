    <script>
        /* 
           VISUAL ENGINE
           Handles background grid, sun, and stars.
           Handles note spawning logic for the CSS highway.
        */

        const canvas = document.getElementById('bg-canvas');
        const ctx = canvas.getContext('2d');
        let width, height;
        let time = 0;

        // Resize Handler
        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        // Colors
        const COL_BG_TOP = '#050011';
        const COL_BG_BOT = '#1a0033';
        const COL_SUN_TOP = '#ff00ff';
        const COL_SUN_BOT = '#ffff00';
        const COL_GRID = '#ff00ff';

        function drawBackground() {
            // 1. Clear & Sky Gradient
            ctx.fillStyle = COL_BG_TOP;
            ctx.fillRect(0, 0, width, height);
            
            const grad = ctx.createLinearGradient(0, 0, 0, height);
            grad.addColorStop(0, COL_BG_TOP);
            grad.addColorStop(0.4, COL_BG_BOT);
            grad.addColorStop(1, '#2a004d');
            ctx.fillStyle = grad;
            ctx.fillRect(0, 0, width, height);

            // 2. Stars (Pixelated)
            ctx.fillStyle = 'white';
            for(let i=0; i<80; i++) {
                const sx = (Math.sin(i * 132.1) * 0.5 + 0.5) * width;
                const sy = (Math.cos(i * 45.3) * 0.5 + 0.5) * (height * 0.45);
                const size = (i % 3 === 0) ? 2 : 1;
                const alpha = 0.5 + Math.sin(time * 0.05 + i) * 0.5;
                ctx.globalAlpha = alpha;
                ctx.fillRect(Math.floor(sx), Math.floor(sy), size, size);
            }
            ctx.globalAlpha = 1.0;

            // 3. Synthwave Sun
            const cx = width / 2;
            const cy = height * 0.42;
            const r = Math.min(width, height) * 0.18;

            // Sun Glow (Subtle)
            ctx.shadowBlur = 20;
            ctx.shadowColor = COL_SUN_TOP;

            const sunGrad = ctx.createLinearGradient(cx, cy - r, cx, cy + r);
            sunGrad.addColorStop(0, COL_SUN_TOP);
            sunGrad.addColorStop(1, COL_SUN_BOT);

            ctx.save();
            ctx.translate(cx, cy);
            ctx.beginPath();
            ctx.arc(0, 0, r, 0, Math.PI * 2);
            ctx.fillStyle = sunGrad;
            ctx.fill();
            
            // Sun Stripes (Cutouts)
            ctx.globalCompositeOperation = 'destination-out';
            ctx.shadowBlur = 0; // No shadow on cutouts
            let stripeOffset = (time * 0.5) % 10;
            for(let i=0; i<12; i++) {
                let y = (i * 8) - r + stripeOffset;
                if (y > r) break;
                let h = i * 1.2 + 2;
                // Only draw if within circle bounds roughly
                ctx.fillRect(-r, y, r*2, h);
            }
            ctx.restore();

            // 4. Perspective Grid
            const horizonY = height * 0.55;
            const vanishX = width / 2;
            
            ctx.strokeStyle = COL_GRID;
            ctx.lineWidth = 2;
            ctx.shadowBlur = 5; // Slight glow on grid lines
            ctx.shadowColor = COL_GRID;
            ctx.globalAlpha = 0.4;

            ctx.beginPath();

            // Verticals
            for (let i = -12; i <= 12; i++) {
                let x1 = vanishX + (i * 15);
                let x2 = vanishX + (i * width * 0.15);
                ctx.moveTo(x1, horizonY);
                ctx.lineTo(x2, height);
            }

            // Horizontals (Moving)
            // Perspective formula: y = h + (depth / z)
            // We animate z by subtracting time
            const speed = 2;
            const gridSpacing = 40;
            
            // Calculate offset for animation
            const offset = (time * speed) % gridSpacing;

            for (let i = 0; i < 25; i++) {
                // Z distance from camera
                let z = (i * gridSpacing) - offset;
                if (z <= 10) continue; // Don't draw behind camera/too close

                // Project Z to Y
                // Simple perspective projection
                let scale = 300 / z;
                let y = horizonY + (scale * 200);

                if (y > height) continue;

                // Fade out near horizon
                let alpha = Math.min(1, (y - horizonY) / (height * 0.4));
                ctx.globalAlpha = alpha * 0.5;

                ctx.moveTo(0, y);
                ctx.lineTo(width, y);
            }
            ctx.stroke();
        }

        /* --- NOTE SPAWNER --- */
        const lanes = [
            document.getElementById('l1'),
            document.getElementById('l2'),
            document.getElementById('l3'),
            document.getElementById('l4')
        ];
        
        // Colors corresponding to lanes
        const noteColors = ['#ff00ff', '#00ffff', '#ffff00', '#ff00ff'];

        function spawnNote() {
            // Random lane
            const laneIdx = Math.floor(Math.random() * 4);
            const lane = lanes[laneIdx];
            
            const note = document.createElement('div');
            note.className = 'note';
            note.style.backgroundColor = noteColors[laneIdx];
            note.style.color = noteColors[laneIdx]; // for shadow
            
            // Animation
            // Using Web Animations API for cleaner cleanup than CSS classes
            const duration = 1500 + Math.random() * 500; // 1.5s to 2s
            
            note.animate([
                { top: '-30px' },
                { top: '100%' }
            ], {
                duration: duration,
                easing: 'linear',
                fill: 'forwards'
            }).onfinish = () => {
                note.remove();
            };

            lane.appendChild(note);
        }

        /* --- MAIN LOOP --- */
        function loop() {
            time++;
            drawBackground();

            // Random note generation
            // Lower probability = fewer notes, cleaner look
            if (Math.random() < 0.04) {
                spawnNote();
            }

            requestAnimationFrame(loop);
        }

        function startGame() {
            // Visual feedback for click
            document.body.style.cursor = 'wait';
            setTimeout(() => {
                document.body.style.cursor = 'default';
                alert("INITIALIZING SEQUENCE...");
            }, 100);
        }

        // Boot
        loop();

    </script>
