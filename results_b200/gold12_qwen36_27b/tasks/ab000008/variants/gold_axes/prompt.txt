    <script>
        /**
         * VISUAL ENGINE
         * Draws the Synthwave Grid and Sun on Canvas
         * Handles the falling notes logic in the CSS-based highway
         */

        const canvas = document.getElementById('bg-canvas');
        const ctx = canvas.getContext('2d');
        let width, height;

        // State
        let time = 0;
        let sunY = 0;

        // Resize handler
        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            sunY = height * 0.4;
        }
        window.addEventListener('resize', resize);
        resize();

        // Colors
        const COL_BG_TOP = '#050011';
        const COL_BG_BOT = '#1a0033';
        const COL_SUN_TOP = '#ff00ff';
        const COL_SUN_BOT = '#ffff00';
        const COL_GRID = '#ff00ff';

        function drawBackground() {
            // Clear
            ctx.fillStyle = COL_BG_TOP;
            ctx.fillRect(0, 0, width, height);

            // Gradient Sky
            const grad = ctx.createLinearGradient(0, 0, 0, height);
            grad.addColorStop(0, COL_BG_TOP);
            grad.addColorStop(0.5, COL_BG_BOT);
            grad.addColorStop(1, '#330055');
            ctx.fillStyle = grad;
            ctx.fillRect(0, 0, width, height);

            // Draw Sun
            const cx = width / 2;
            const cy = height * 0.35;
            const r = Math.min(width, height) * 0.2;

            const sunGrad = ctx.createLinearGradient(cx, cy - r, cx, cy + r);
            sunGrad.addColorStop(0, COL_SUN_TOP);
            sunGrad.addColorStop(1, COL_SUN_BOT);

            ctx.save();
            ctx.translate(cx, cy);
            
            // Sun Glow
            ctx.shadowBlur = 40;
            ctx.shadowColor = COL_SUN_TOP;
            
            // Draw Sun with horizontal stripes (The classic Synthwave look)
            ctx.beginPath();
            ctx.arc(0, 0, r, 0, Math.PI * 2);
            ctx.fillStyle = sunGrad;
            ctx.fill();
            
            // Cutout stripes
            ctx.globalCompositeOperation = 'destination-out';
            for(let i=0; i<10; i++) {
                let y = (i * r / 5) - r;
                let h = i * 1.5 + 2;
                ctx.fillRect(-r, y + r*0.4, r*2, h);
            }
            ctx.restore();

            // Draw Stars
            ctx.fillStyle = 'white';
            ctx.shadowBlur = 0;
            for(let i=0; i<50; i++) {
                // Pseudo-random stars based on index
                const sx = (Math.sin(i * 123.45) * 0.5 + 0.5) * width;
                const sy = (Math.cos(i * 67.89) * 0.5 + 0.5) * (height * 0.4);
                const size = (i % 3 === 0) ? 2 : 1;
                // Twinkle
                const alpha = Math.abs(Math.sin(time * 0.05 + i));
                ctx.globalAlpha = alpha;
                ctx.fillRect(sx, sy, size, size);
            }
            ctx.globalAlpha = 1.0;

            // Draw Perspective Grid
            const horizonY = height * 0.5;
            const vanishX = width / 2;
            const gridSpeed = (time * 2) % 40; // Speed of grid moving

            ctx.strokeStyle = COL_GRID;
            ctx.lineWidth = 2;
            ctx.shadowBlur = 10;
            ctx.shadowColor = COL_GRID;
            
            ctx.beginPath();

            // Vertical lines (Perspective)
            for (let i = -10; i <= 10; i++) {
                // Fan out from center
                let x1 = vanishX + (i * 20); 
                let x2 = vanishX + (i * width * 0.2); 
                ctx.moveTo(x1, horizonY);
                ctx.lineTo(x2, height);
            }

            // Horizontal lines (Moving towards camera)
            // We simulate Z-depth with non-linear spacing
            for (let i = 0; i < 20; i++) {
                // Exponential spacing for perspective
                let z = i * 20 - gridSpeed; 
                if (z <= 0) continue;
                
                // Perspective projection math simplified
                // y = horizon + (height / z_factor)
                // Using a simple perspective curve
                let perspectiveScale = (z / 200); 
                if (perspectiveScale > 1) continue;
                
                let y = horizonY + (height * 0.6) * (1 - (1 / (z * 0.05 + 1)));
                
                // Adjust for screen
                y = horizonY + (height - horizonY) * (1 - (100 / (z + 100)));
                
                if (y > horizonY && y < height) {
                    ctx.moveTo(0, y);
                    ctx.lineTo(width, y);
                }
            }
            ctx.stroke();
        }

        // --- NOTE SPAWNER LOGIC ---
        // Spawns visual notes on the HTML highway
        const highway = document.getElementById('highway');
        const lanes = [
            document.getElementById('l1'),
            document.getElementById('l2'),
            document.getElementById('l3'),
            document.getElementById('l4')
        ];
        const colors = ['#ff00ff', '#00ffff', '#ffff00', '#ff00ff'];

        function spawnNote() {
            const laneIdx = Math.floor(Math.random() * 4);
            const lane = lanes[laneIdx];
            
            const note = document.createElement('div');
            note.className = 'note';
            note.style.backgroundColor = colors[laneIdx];
            note.style.color = colors[laneIdx]; // for box-shadow
            note.style.top = '-20px';
            
            // Randomize speed slightly
            const duration = 1 + Math.random() * 0.5;
            note.style.transition = `top ${duration}s linear`;
            
            lane.appendChild(note);

            // Trigger animation next frame
            requestAnimationFrame(() => {
                note.style.top = '100%';
            });

            // Cleanup
            setTimeout(() => {
                note.remove();
            }, duration * 1000);
        }

        // Main Loop
        function loop() {
            time++;
            drawBackground();
            
            // Random note spawning
            if (Math.random() < 0.05) {
                spawnNote();
            }

            requestAnimationFrame(loop);
        }

        // Start
        loop();
