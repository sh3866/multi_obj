<script>
/**
 * Spherical Conway's Game of Life
 * 
 * Design Improvements:
 * 1. Four-hue luminous palette (Cyan, Magenta, Purple, Gold).
 * 2. Opacity and scale based on Z-depth for spherical volume.
 * 3. Limb darkening (fading at edges).
 * 4. Compact left panel with better typography.
 */

// --- Configuration ---
const CONFIG = {
  cellCount: 4000, // Increased density
  sphereRadius: 280,
  baseCellSize: 2.5,
  rotationSpeed: 0.001,
  // Palette: Cyan -> Magenta -> Purple -> Gold
  palette: [
    { h: 180, s: 100, l: 60 }, // Cyan
    { h: 320, s: 100, l: 60 }, // Magenta
    { h: 260, s: 80, l: 65 },  // Purple
    { h: 40, s: 100, l: 60 }   // Gold
  ]
};

// --- State ---
let state = {
  cells: [],
  generation: 0,
  isRunning: true,
  speed: 15,
  lastFrameTime: 0,
  rotation: { x: 0.2, y: 0.4 }, // Slight initial tilt
  zoom: 1,
  isDragging: false,
  lastMouse: { x: 0, y: 0 },
  mouseVelocity: { x: 0, y: 0 }
};

// --- Canvas Setup ---
const canvas = document.getElementById('sphereCanvas');
const ctx = canvas.getContext('2d', { alpha: false });

let width, height;

function resize() {
  width = window.innerWidth;
  height = window.innerHeight;
  canvas.width = width;
  canvas.height = height;
}
window.addEventListener('resize', resize);
resize();

// --- Math Helpers ---
const Vec3 = {
  dist: (v1, v2) => {
    const dx = v1.x - v2.x;
    const dy = v1.y - v2.y;
    const dz = v1.z - v2.z;
    return Math.sqrt(dx*dx + dy*dy + dz*dz);
  }
};

// Fibonacci Lattice Sphere Generation
function createSpherePoints(count, radius) {
  const points = [];
  const goldenAngle = Math.PI * (3 - Math.sqrt(5));
  
  for (let i = 0; i < count; i++) {
    const y = 1 - (i / (count - 1)) * 2;
    const radiusAtY = Math.sqrt(1 - y * y);
    const theta = goldenAngle * i;
    
    points.push({
      x: Math.cos(theta) * radiusAtY * radius,
      y: y * radius,
      z: Math.sin(theta) * radiusAtY * radius
    });
  }
  return points;
}

// --- Game Logic ---

function initGame() {
  const points = createSpherePoints(CONFIG.cellCount, CONFIG.sphereRadius);
  
  // Spatial Hashing for neighbors (Grid approach mapped to sphere)
  // To keep it simple and performant, we use a distance threshold relative to density
  const threshold = (2 * Math.PI * CONFIG.sphereRadius) / Math.sqrt(CONFIG.cellCount) * 1.1; 
  
  state.cells = points.map((p, i) => {
    const neighbors = [];
    // Optimization: Only check a subset or use a grid if N is huge. 
    // For 4000 points, O(N^2) is roughly 16M checks. It's heavy for JS loop.
    // Better approach: Check points with index difference within a range? No, indices aren't spatial.
    // We will do a brute force but limit checks or optimize slightly.
    
    // Optimized neighbor finding:
    // Since it's a sphere, we can't easily grid it.
    // Let's just do the distance check but break early? No.
    // We accept the cost for init only.
    for (let j = 0; j < points.length; j++) {
      if (i === j) continue;
      if (Vec3.dist(p, points[j]) < threshold) {
        neighbors.push(j);
      }
    }
    
    return {
      pos: p,
      neighbors: neighbors,
      alive: Math.random() > 0.8, // 20% density
      bornGen: 0,
      age: 0
    };
  });
  
  state.generation = 0;
  updateStats();
}

function stepGame() {
  const nextStates = [];
  
  // Parallel update logic
  for(let i=0; i<state.cells.length; i++) {
    const cell = state.cells[i];
    let neighborCount = 0;
    
    for(let k=0; k<cell.neighbors.length; k++) {
      if(state.cells[cell.neighbors[k]].alive) neighborCount++;
    }
    
    let nextAlive = false;
    let isBorn = false;

    if (cell.alive) {
      if (neighborCount === 2 || neighborCount === 3) nextAlive = true;
    } else {
      if (neighborCount === 3) {
        nextAlive = true;
        isBorn = true;
      }
    }
    
    nextStates.push({
      alive: nextAlive,
      bornGen: isBorn ? state.generation : cell.bornGen,
      age: nextAlive ? (cell.alive ? cell.age + 1 : 0) : 0
    });
  }

  // Apply
  for(let i=0; i<state.cells.length; i++) {
    const s = nextStates[i];
    state.cells[i].alive = s.alive;
    state.cells[i].bornGen = s.bornGen;
    state.cells[i].age = s.age;
  }

  state.generation++;
  updateStats();
}

// --- Rendering ---

function project(p, rotX, rotY, zoom) {
  // Rotate Y
  let x = p.x * Math.cos(rotY) - p.z * Math.sin(rotY);
  let z = p.z * Math.cos(rotY) + p.x * Math.sin(rotY);
  
  // Rotate X
  let y = p.y * Math.cos(rotX) - z * Math.sin(rotX);
  let zFinal = z * Math.cos(rotX) + p.y * Math.sin(rotX);
  
  // Perspective
  // Camera position roughly at z = -600
  const fov = 800 * zoom;
  const dist = 600; // Camera distance
  const scale = fov / (dist - zFinal); // Standard perspective projection
  
  return {
    x: x * scale + width / 2,
    y: y * scale + height / 2,
    scale: scale,
    z: zFinal,
    depthNorm: (zFinal / CONFIG.sphereRadius) // -1 (front) to 1 (back)
  };
}

function draw() {
  // Clear
  ctx.fillStyle = '#000000';
  ctx.fillRect(0, 0, width, height);

  // Pre-calculate projection for all cells
  const projectedCells = state.cells.map(cell => {
    const proj = project(cell.pos, state.rotation.x, state.rotation.y, state.zoom);
    return { cell, proj };
  });
  
  // Sort back-to-front (Painter's Algorithm)
  projectedCells.sort((a, b) => b.proj.z - a.proj.z);

  // Draw Sphere Atmosphere (Glow)
  const cx = width / 2;
  const cy = height / 2;
  const r = CONFIG.sphereRadius * state.zoom * (800 / (600)); // Approximate screen radius
  
  const grad = ctx.createRadialGradient(cx, cy, r * 0.8, cx, cy, r * 1.4);
  grad.addColorStop(0, 'rgba(0, 243, 255, 0)');
  grad.addColorStop(0.6, 'rgba(0, 243, 255, 0.03)');
  grad.addColorStop(1, 'rgba(0, 0, 0, 0)');
  
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.arc(cx, cy, r * 1.4, 0, Math.PI * 2);
  ctx.fill();

  // Draw Cells
  projectedCells.forEach(({ cell, proj }) => {
    // Culling: Don't draw if too far back (optional, but looks cleaner if we see "through" slightly)
    // We keep drawing back faces but very dim.
    
    // Calculate visual properties based on depth
    // proj.depthNorm: -1 is closest, 1 is furthest
    // We want front to be bright, back to be dim.
    const depthFactor = (proj.depthNorm + 1) / 2; // 0 (back) to 1 (front)
    
    // Limb darkening effect (fade at edges of sphere)
    // We can approximate this by checking if x/y is near the edge of the circle r
    // But Z-depth sorting handles occlusion.
    
    const size = CONFIG.baseCellSize * proj.scale;
    
    if (cell.alive) {
      // Color Logic: 4 hues based on age/generation
      const hueIndex = Math.floor(cell.bornGen / 10) % CONFIG.palette.length;
      const color = CONFIG.palette[hueIndex];
      
      // Brightness based on depth (front is brighter)
      // Base opacity 0.3, max 1.0
      const alpha = 0.2 + (depthFactor * 0.8);
      const lightness = color.l * (0.5 + depthFactor * 0.5); // Darker in back
      
      ctx.fillStyle = `hsla(${color.h}, ${color.s}%, ${lightness}%, ${alpha})`;
      
      // Glow effect for front cells
      if (depthFactor > 0.8 && size > 3) {
        ctx.shadowBlur = 8 * proj.scale;
        ctx.shadowColor = `hsla(${color.h}, ${color.s}%, ${color.l}%, 0.6)`;
      } else {
        ctx.shadowBlur = 0;
      }
    } else {
      // Dead cells: barely visible grid
      const alpha = 0.05 + (depthFactor * 0.1);
      ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
      ctx.shadowBlur = 0;
    }
    
    ctx.beginPath();
    ctx.arc(proj.x, proj.y, size, 0, Math.PI * 2);
    ctx.fill();
  });
  
  ctx.shadowBlur = 0; // Reset
}

// --- Main Loop ---

function loop(timestamp) {
  if (state.isRunning) {
    // Auto rotation (slows down if dragging)
    state.rotation.y += CONFIG.rotationSpeed;
    
    // Update Logic based on speed
    // Use modulo to control updates per frame if speed is high, or simple timer
    const interval = 1000 / state.speed;
    if (timestamp - state.lastFrameTime > interval) {
      stepGame();
      state.lastFrameTime = timestamp;
    }
  } else {
    // Still rotate slightly when paused for effect
    state.rotation.y += CONFIG.rotationSpeed * 0.2;
  }
  
  draw();
  requestAnimationFrame(loop);
}

// --- Interaction ---

function updateStats() {
  document.getElementById('gen-count').innerText = state.generation;
  const pop = state.cells.filter(c => c.alive).length;
  document.getElementById('pop-count').innerText = pop;
}

// Controls
document.getElementById('btn-pause').addEventListener('click', (e) => {
  state.isRunning = !state.isRunning;
  e.target.innerText = state.isRunning ? "Pause" : "Resume";
  e.target.classList.toggle('primary');
  e.target.classList.toggle('secondary');
});

document.getElementById('btn-reset').addEventListener('click', () => {
  initGame();
});

document.getElementById('btn-step').addEventListener('click', () => {
  state.isRunning = false;
  document.getElementById('btn-pause').innerText = "Resume";
  document.getElementById('btn-pause').classList.remove('primary');
  document.getElementById('btn-pause').classList.add('secondary');
  stepGame();
  draw();
});

document.getElementById('speed-slider').addEventListener('input', (e) => {
  state.speed = parseInt(e.target.value);
});

// Mouse Drag Rotation
const container = document.getElementById('canvas-container');

container.addEventListener('mousedown', (e) => {
  state.isDragging = true;
  state.lastMouse = { x: e.clientX, y: e.clientY };
});

window.addEventListener('mousemove', (e) => {
  if (!state.isDragging) return;
  const deltaX = e.clientX - state.lastMouse.x;
  const deltaY = e.clientY - state.lastMouse.y;
  
  state.rotation.y += deltaX * 0.005;
  state.rotation.x += deltaY * 0.005;
  
  state.lastMouse = { x: e.clientX, y: e.clientY };
});

window.addEventListener('mouseup', () => {
  state.isDragging = false;
});

// Zoom
container.addEventListener('wheel', (e) => {
  e.preventDefault();
  const zoomSensitivity = 0.001;
  state.zoom -= e.deltaY * zoomSensitivity;
  state.zoom = Math.max(0.5, Math.min(3, state.zoom));
});

// --- Init ---
initGame();
requestAnimationFrame(loop);

</script>
