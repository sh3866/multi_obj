<script>
  /**
   * LUMINA - Game Logic & Rendering
   * Optimized for visual depth and "Cinematic" feel.
   */
  
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  let mouse = { x: -1000, y: -1000 };
  let isMouseDown = false;
  let time = 0;

  // Configuration
  const COLORS = {
    cyan: '#00f0ff',
    magenta: '#ff0055',
    lime: '#ccff00',
    purple: '#bd00ff'
  };
  
  // Resize Handling
  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
  }
  window.addEventListener('resize', resize);
  resize();

  // Input Handling
  window.addEventListener('mousemove', e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
  });
  window.addEventListener('mousedown', () => isMouseDown = true);
  window.addEventListener('mouseup', () => {
    isMouseDown = false;
    checkCatch();
  });
  // Touch support
  window.addEventListener('touchmove', e => {
    e.preventDefault();
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
    isMouseDown = true;
  });
  window.addEventListener('touchend', () => {
    isMouseDown = false;
    checkCatch();
  });

  // --- Classes ---

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 4 + 2;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.02 + 0.015;
      this.size = Math.random() * 4 + 2;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.vx *= 0.95; // Friction
      this.vy *= 0.95;
      this.life -= this.decay;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.shadowBlur = 10;
      ctx.shadowColor = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor(zIndex) {
      this.zIndex = zIndex; // 0 = far, 1 = mid, 2 = near
      this.reset();
      // Initial random placement
      this.x = Math.random() * width;
      this.y = Math.random() * height;
    }

    reset() {
      // Spawn logic based on depth
      const spawnSide = Math.random() < 0.5 ? -100 : width + 100;
      this.x = spawnSide;
      
      // Depth affects Y position preference (lower = more likely near)
      const yOffset = this.zIndex === 0 ? -100 : (this.zIndex === 2 ? 100 : 0);
      this.y = (Math.random() * (height * 0.7)) + yOffset;

      // Speed scales with depth (far = slow, near = fast)
      const speedMult = this.zIndex === 0 ? 0.5 : (this.zIndex === 2 ? 1.5 : 1);
      this.vx = (Math.random() - 0.5) * 2 * speedMult;
      this.vy = (Math.random() - 0.5) * 1.5 * speedMult;
      
      // Size scales with depth
      const baseSize = this.zIndex === 0 ? 15 : (this.zIndex === 2 ? 35 : 22);
      this.radius = baseSize + (Math.random() * 10);
      
      // Color assignment
      if (this.zIndex === 0) this.color = COLORS.cyan;
      else if (this.zIndex === 1) this.color = COLORS.magenta;
      else this.color = COLORS.lime;

      this.angle = 0;
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = (Math.random() * 0.05 + 0.02) * speedMult;
      this.isCaught = false;
    }

    update() {
      if (this.isCaught) return;

      // Movement
      this.x += this.vx;
      this.y += this.vy;
      this.wobble += this.wobbleSpeed;
      
      // Organic sine wave drift
      this.y += Math.sin(this.wobble) * 0.5;

      // Boundary checks
      if (this.x < -150) this.vx = Math.abs(this.vx);
      if (this.x > width + 150) this.vx = -Math.abs(this.vx);
      if (this.y < -50) this.vy = Math.abs(this.vy);
      if (this.y > height + 50) this.vy = -Math.abs(this.vy);

      // Avoid Mouse (Net) - Reaction speed depends on depth
      const dx = this.x - mouse.x;
      const dy = this.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      // Detection radius
      const detectRange = 180 + (this.zIndex * 50);

      if (dist < detectRange) {
        // Flee behavior
        const force = (detectRange - dist) / detectRange;
        const fleeStrength = 0.8 + (this.zIndex * 0.3);
        this.vx += (dx / dist) * force * fleeStrength;
        this.vy += (dy / dist) * force * fleeStrength;
      }

      // Speed limiter
      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      const maxSpeed = 4 + (this.zIndex * 1.5);
      if (speed > maxSpeed) {
        this.vx = (this.vx / speed) * maxSpeed;
        this.vy = (this.vy / speed) * maxSpeed;
      }
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      
      // Glow intensity based on depth
      const glowAmount = 15 + (this.zIndex * 10);
      ctx.shadowBlur = glowAmount;
      ctx.shadowColor = this.color;
      
      // Body
      ctx.fillStyle = this.color;
      ctx.beginPath();
      
      // Draw organic shape
      const r = this.radius;
      const wobbleOffset = Math.sin(this.wobble * 2) * (r * 0.15);
      
      ctx.moveTo(r + wobbleOffset, 0);
      ctx.bezierCurveTo(r, r, 0, r, -r - wobbleOffset, 0);
      ctx.bezierCurveTo(-r, -r, 0, -r, r + wobbleOffset, 0);
      
      ctx.fill();

      // Inner Core
      ctx.fillStyle = '#fff';
      ctx.shadowBlur = 0;
      ctx.beginPath();
      ctx.arc(0, 0, r * 0.35, 0, Math.PI * 2);
      ctx.fill();
      
      // Eye/Detail for larger creatures
      if (this.zIndex > 0) {
        ctx.fillStyle = 'rgba(0,0,0,0.3)';
        ctx.beginPath();
        ctx.arc(r * 0.2, -r * 0.1, r * 0.15, 0, Math.PI * 2);
        ctx.fill();
      }

      ctx.restore();
    }
  }

  // --- Game Functions ---

  function init() {
    creatures = [];
    // Create 3 distinct layers of creatures
    // Far layer (Cyan, slow, small)
    for(let i=0; i<4; i++) creatures.push(new Creature(0));
    // Mid layer (Magenta, medium)
    for(let i=0; i<3; i++) creatures.push(new Creature(1));
    // Near layer (Lime, fast, large)
    for(let i=0; i<2; i++) creatures.push(new Creature(2));
    
    // Sort by Z index so near creatures draw on top
    creatures.sort((a, b) => a.zIndex - b.zIndex);

    score = 0;
    updateScoreDisplay();
  }

  function updateScoreDisplay() {
    scoreEl.innerText = score.toString().padStart(2, '0');
  }

  function checkCatch() {
    // Capture zone size
    const catchRadius = 70; 

    // Iterate backwards to catch top-most creatures first
    for (let i = creatures.length - 1; i >= 0; i--) {
      let c = creatures[i];
      if (c.isCaught) continue;
      
      const dx = c.x - mouse.x;
      const dy = c.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);

      if (dist < catchRadius) {
        catchCreature(c);
        break; // Catch one at a time
      }
    }
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    updateScoreDisplay();
    
    // Visual FX - Explosion
    for(let i=0; i<20; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }

    // Show Toast
    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1200);

    // Respawn logic
    setTimeout(() => {
      creature.reset();
    }, 3000);
  }

  function resetGame() {
    init();
  }

  function drawNet() {
    // Draw the player's net/cursor
    // Position slightly offset to simulate a "hand holding a net" feel
    const netX = mouse.x;
    const netY = mouse.y;

    ctx.save();
    ctx.translate(netX, netY);
    
    // Rim Light Effect
    ctx.shadowBlur = 20;
    ctx.shadowColor = COLORS.cyan;
    
    // Net Shape
    ctx.strokeStyle = isMouseDown ? '#fff' : COLORS.cyan;
    ctx.lineWidth = 2;
    
    // Main Ring
    ctx.beginPath();
    ctx.arc(0, 0, 65, 0, Math.PI * 2);
    ctx.stroke();
    
    // Inner Tech Ring
    ctx.strokeStyle = `rgba(0, 240, 255, ${isMouseDown ? 0.8 : 0.3})`;
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.arc(0, 0, 50, 0, Math.PI * 2);
    ctx.stroke();
    ctx.setLineDash([]);

    // Crosshair
    ctx.beginPath();
    ctx.moveTo(-15, 0); ctx.lineTo(-5, 0);
    ctx.moveTo(5, 0); ctx.lineTo(15, 0);
    ctx.moveTo(0, -15); ctx.lineTo(0, -5);
    ctx.moveTo(0, 5); ctx.lineTo(0, 15);
    ctx.stroke();

    // Capture "Sucking" effect if held down
    if (isMouseDown) {
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.arc(0, 0, 85, 0, Math.PI * 2);
      ctx.stroke();
    }

    ctx.restore();
  }

  function loop() {
    time += 0.01;
    
    // Clear
    ctx.clearRect(0, 0, width, height);

    // Draw Particles (Behind creatures)
    for (let i = particles.length - 1; i >= 0; i--) {
      let p = particles[i];
      p.update();
      p.draw(ctx);
      if (p.life <= 0) particles.splice(i, 1);
    }

    // Update & Draw Creatures (Sorted by Z)
    creatures.forEach(c => {
      c.update();
      c.draw(ctx);
    });

    // Draw Player Net (Always on top of creatures)
    drawNet();

    requestAnimationFrame(loop);
  }

  // Start
  init();
  loop();

</script>
