<script>
  /**
   * Game Logic & Rendering
   * Advanced Canvas implementation for organic visuals.
   */
  
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d', { alpha: true });
  const scoreEl = document.getElementById('scoreDisplay');
  const toastEl = document.getElementById('toast');

  let width, height;
  let score = 0;
  let particles = [];
  let creatures = [];
  let player = { x: 0, y: 0, angle: 0 };
  let mouse = { x: -1000, y: -1000 };
  let isMouseDown = false;
  let time = 0;

  // Configuration
  const COLORS = ['#00f0ff', '#ff0055', '#ccff00', '#bd00ff'];
  const CREATURE_COUNT = 5;

  // Resize Handling
  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
    // Keep player in bounds
    if (player.x > width) player.x = width - 50;
    if (player.y > height) player.y = height - 50;
  }
  window.addEventListener('resize', resize);
  resize();

  // Input Handling
  window.addEventListener('mousemove', e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
  });
  window.addEventListener('mousedown', () => isMouseDown = true);
  window.addEventListener('mouseup', () => {
    isMouseDown = false;
    checkCatch();
  });
  // Touch support
  window.addEventListener('touchmove', e => {
    e.preventDefault();
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
  }, { passive: false });
  window.addEventListener('touchstart', e => {
    mouse.x = e.touches[0].clientX;
    mouse.y = e.touches[0].clientY;
    isMouseDown = true;
  });
  window.addEventListener('touchend', () => {
    isMouseDown = false;
    checkCatch();
  });

  // --- Classes ---

  class Particle {
    constructor(x, y, color) {
      this.x = x;
      this.y = y;
      this.color = color;
      const angle = Math.random() * Math.PI * 2;
      const speed = Math.random() * 4 + 2;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.life = 1.0;
      this.decay = Math.random() * 0.02 + 0.01;
      this.size = Math.random() * 4 + 2;
    }

    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.vx *= 0.95; // Friction
      this.vy *= 0.95;
      this.life -= this.decay;
    }

    draw(ctx) {
      ctx.globalAlpha = this.life;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1.0;
    }
  }

  class Creature {
    constructor() {
      this.reset(true);
    }

    reset(initial = false) {
      // Spawn logic
      if (initial) {
        this.x = Math.random() * width;
        this.y = Math.random() * (height * 0.8); // Keep away from very bottom
      } else {
        this.x = Math.random() < 0.5 ? -50 : width + 50;
        this.y = Math.random() * (height * 0.7);
      }
      
      this.vx = (Math.random() - 0.5) * 1.5;
      this.vy = (Math.random() - 0.5) * 1.5;
      this.radius = Math.random() * 15 + 10;
      this.color = COLORS[Math.floor(Math.random() * COLORS.length)];
      this.wobble = Math.random() * Math.PI * 2;
      this.wobbleSpeed = Math.random() * 0.03 + 0.01;
      this.isCaught = false;
      this.depth = Math.random(); // For parallax feel
    }

    update() {
      if (this.isCaught) return;

      // Movement
      this.x += this.vx;
      this.y += this.vy;
      this.wobble += this.wobbleSpeed;
      
      // Organic drift
      this.x += Math.sin(this.wobble) * 0.3;

      // Wall Bounce
      if (this.x < 0) { this.x = 0; this.vx *= -1; }
      if (this.x > width) { this.x = width; this.vx *= -1; }
      if (this.y < 0) { this.y = 0; this.vy *= -1; }
      if (this.y > height - 100) { this.y = height - 100; this.vy *= -1; } // Stay above terrain

      // Evasion Logic (AI)
      const dx = this.x - player.x;
      const dy = this.y - player.y;
      const dist = Math.sqrt(dx*dx + dy*dy);
      
      // Panic radius
      if (dist < 180) {
        const force = (180 - dist) / 180;
        // Flee direction
        let fleeX = (dx / dist);
        let fleeY = (dy / dist);
        
        // Add some jitter to make it unpredictable
        fleeX += (Math.random() - 0.5) * 0.5;
        fleeY += (Math.random() - 0.5) * 0.5;

        this.vx += fleeX * force * 0.8;
        this.vy += fleeY * force * 0.8;
      }

      // Speed Cap
      const speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
      if (speed > 3.5) {
        this.vx = (this.vx / speed) * 3.5;
        this.vy = (this.vy / speed) * 3.5;
      }
    }

    draw(ctx) {
      if (this.isCaught) return;

      ctx.save();
      ctx.translate(this.x, this.y);
      
      // Depth scaling
      const scale = 0.8 + (this.depth * 0.4);
      ctx.scale(scale, scale);
      
      // Glow
      ctx.shadowBlur = 25;
      ctx.shadowColor = this.color;
      
      // Draw distinct shape based on type (randomized on creation, but here we use color for variety)
      ctx.fillStyle = this.color;
      
      // Pulsating effect
      const pulse = 1 + Math.sin(time * 0.05 + this.wobble) * 0.1;
      ctx.scale(pulse, pulse);

      ctx.beginPath();
      // Create a jellyfish/blob shape
      const r = this.radius;
      ctx.moveTo(r, 0);
      for (let i = 1; i <= 6; i++) {
        const angle = (i / 6) * Math.PI * 2;
        const wobbleR = r + Math.sin(this.wobble * 2 + i) * 3;
        ctx.lineTo(Math.cos(angle) * wobbleR, Math.sin(angle) * wobbleR);
      }
      ctx.closePath();
      ctx.fill();

      // Inner Core (Eye)
      ctx.shadowBlur = 0;
      ctx.fillStyle = '#fff';
      ctx.beginPath();
      ctx.arc(r * 0.2, 0, r * 0.25, 0, Math.PI * 2);
      ctx.fill();

      ctx.restore();
    }
  }

  // --- Player Drawing ---

  function drawPlayer() {
    // Player follows mouse with slight lag (handled by just drawing at mouse pos for responsiveness in this simple version)
    // Or we could lerp player.x/y to mouse.x/y for weight.
    // Let's just draw at mouse for tight control.
    
    const px = player.x;
    const py = player.y;

    ctx.save();
    ctx.translate(px, py);
    
    // Calculate angle towards mouse movement or just face right/down
    // For this game, let's make the player face the direction they are moving
    // (Simplified: just draw a silhouette)
    
    // 1. The Silhouette (Person)
    ctx.fillStyle = '#000'; // Pure black silhouette
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 4;
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';

    // Body
    ctx.beginPath();
    ctx.moveTo(-10, 10); // Feet
    ctx.lineTo(-10, -15); // Torso
    ctx.lineTo(5, -25); // Head/Shoulder
    ctx.lineTo(15, -15); // Back
    ctx.lineTo(5, 10);
    ctx.fill();

    // Head
    ctx.beginPath();
    ctx.arc(2, -30, 6, 0, Math.PI*2);
    ctx.fill();

    // 2. The Net
    // Net is held in the "right" hand (positive X relative to body)
    // We want the net to be at the mouse cursor position relative to the player
    // But wait, the player IS the mouse cursor in this implementation logic.
    // Let's adjust: The "Player" is the cursor.
    
    // Net Handle
    ctx.strokeStyle = '#111';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(0, -10);
    ctx.lineTo(30, -30); // Handle extending out
    ctx.stroke();

    // Net Bag (Mesh)
    ctx.translate(30, -30); // Move to end of handle
    ctx.rotate(Math.PI / 4); // Angle it down slightly
    
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.3)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.ellipse(0, 0, 25, 35, 0, 0, Math.PI * 2);
    ctx.stroke();
    
    // Net Grid
    ctx.beginPath();
    for(let i=-20; i<=20; i+=10) {
        ctx.moveTo(i, -30); ctx.lineTo(i, 30);
    }
    for(let i=-30; i<=30; i+=10) {
        ctx.moveTo(-25, i); ctx.lineTo(25, i);
    }
    ctx.stroke();

    // Net Glow (The active area)
    ctx.shadowBlur = 15;
    ctx.shadowColor = '#00f0ff';
    ctx.strokeStyle = '#00f0ff';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.ellipse(0, 0, 25, 35, 0, 0, Math.PI * 2);
    ctx.stroke();

    // Capture Indicator (Pulsing circle in center of net)
    ctx.shadowBlur = 10;
    ctx.fillStyle = isMouseDown ? 'rgba(255, 255, 255, 0.4)' : 'rgba(0, 240, 255, 0.1)';
    ctx.beginPath();
    ctx.arc(0, 0, 15, 0, Math.PI * 2);
    ctx.fill();

    ctx.restore();
  }

  // --- Game Functions ---

  function init() {
    creatures = [];
    for(let i=0; i<CREATURE_COUNT; i++) {
      creatures.push(new Creature());
    }
    score = 0;
    scoreEl.innerText = score;
    // Center player
    player.x = width / 2;
    player.y = height / 2;
    mouse.x = width / 2;
    mouse.y = height / 2;
  }

  function checkCatch() {
    // Collision detection
    // The "Hitbox" is the center of the net relative to the player (which is at mouse pos)
    // Since drawPlayer translates to player.x/y (mouse.x/y), the net center is offset.
    // Net center offset approx: x+30, y-30 (rotated).
    // Simplified: Hitbox is just a radius around the mouse cursor.
    
    const catchRadius = 60; 

    creatures.forEach(c => {
      if (c.isCaught) return;
      
      const dx = c.x - mouse.x;
      const dy = c.y - mouse.y;
      const dist = Math.sqrt(dx*dx + dy*dy);

      if (dist < catchRadius) {
        catchCreature(c);
      }
    });
  }

  function catchCreature(creature) {
    creature.isCaught = true;
    score++;
    scoreEl.innerText = score;
    
    // Visual FX
    for(let i=0; i<20; i++) {
      particles.push(new Particle(creature.x, creature.y, creature.color));
    }

    // Show Toast
    toastEl.classList.add('active');
    setTimeout(() => toastEl.classList.remove('active'), 1200);

    // Respawn logic
    setTimeout(() => {
      creature.reset();
    }, 3000);
  }

  function resetGame() {
    init();
    // Shake effect or feedback could go here
    toastEl.innerText = "Collection Reset";
    toastEl.classList.add('active');
    setTimeout(() => {
        toastEl.classList.remove('active');
        toastEl.innerText = "Specimen Captured!";
    }, 1000);
  }

  function loop() {
    time++;
    
    // Update Player Position (Mouse Tracking)
    // We add a tiny bit of lag for weight
    player.x += (mouse.x - player.x) * 0.2;
    player.y += (mouse.y - player.y) * 0.2;

    // Clear
    ctx.clearRect(0, 0, width, height);

    // Update & Draw Creatures
    // Sort by Y to give pseudo-depth (draw lower creatures on top)
    creatures.sort((a, b) => a.y - b.y);
    
    creatures.forEach(c => {
      c.update();
      c.draw(ctx);
    });

    // Update & Draw Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      let p = particles[i];
      p.update();
      p.draw(ctx);
      if (p.life <= 0) particles.splice(i, 1);
    }

    // Draw Player (Always on top of creatures, below UI)
    drawPlayer();

    requestAnimationFrame(loop);
  }

  // Start
  init();
  loop();

</script>
